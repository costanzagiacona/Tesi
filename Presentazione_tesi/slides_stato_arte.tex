\section{Panoramica sugli UAV VTOL}

\begin{frame}{Panoramica: Configurazioni Principali}
    La categoria UAV VTOL si articola in tre configurazioni strutturali:
    \begin{itemize}
        \item \textbf{Multirotori:}
        \begin{itemize}
            \item Semplicità meccanica e hovering stabile.
            \item Bassa efficienza energetica nel volo traslato (autonomia limitata).
        \end{itemize}
        \item \textbf{Ala fissa:}
        \begin{itemize}
            \item Elevata efficienza aerodinamica e lunghe percorrenze.
            \item Necessità di spazi o infrastrutture per decollo/atterraggio.
        \end{itemize}
        \item \textbf{Ibridi (Tilting-rotor, Tilting-wing, Tailsitter):}
        \begin{itemize}
            \item Combinano la versatilità dei multirotori con l'efficienza dell'ala fissa.
            \item Capacità di transizione tra volo verticale e orizzontale.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Sfide di Modellazione e Controllo}
    Lo sviluppo di VTOL ibridi presenta sfide significative, concentrate nella \textbf{fase di transizione}:
    \begin{block}{Criticità Dinamiche}
        \begin{itemize}
            \item \textbf{Non linearità aerodinamiche:} Interazione complessa tra flussi dei rotori, ali e fusoliera.
            \item \textbf{Accoppiamento:} Forte legame tra gradi di libertà longitudinali e laterali.
            \item \textbf{Variazione Parametrica:} Il sistema cambia drasticamente tra hovering statico e crociera.
        \end{itemize}
    \end{block}
\end{frame}

% --- SEZIONE 2 ---
\section{UAV VTOL con ali indipendentemente inclinabili}

\begin{frame}{Tilt delle Ali Indipendente [1]}
    \textbf{Obiettivo}: Superare i limiti di manovrabilità dei tricotteri e gestire la coppia del rotore di coda senza rotori ausiliari.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Modellazione 6-DOF}: Il controllo differenziale del \textit{tilt} alare genera una differenza di portanza che induce momenti di rollio.
        \item \textbf{Strategia di Controllo (PID/PI)}:
        \begin{itemize}
            \item \textit{Modalità Verticale}: PID classici su motori e servi per la stabilizzazione.
            \item \textit{Modalità Orizzontale}: Rollio via tilt differenziale; Imbardata tramite minimizzazione dello scivolamento laterale (\textit{sideslip}).
        \end{itemize}
        \item \textbf{Transizione}: Gestita da una logica a stati finiti per un passaggio fluido.
    \end{itemize}
\end{frame}

% --- SEZIONE 3 ---
\section{Modellazione e controllo Backstepping per Quadricotteri}

\begin{frame}{Backstepping Sequenziale [2]}
    Approccio al controllo non lineare per quadricotteri (sistema sottattuato e accoppiato).
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Architettura Ibrida}:
        \begin{enumerate}
            \item \textit{Sottosistema Traslazionale}: Linearizzazione con retroazione + Regolatore PD (Posizione/Altitudine).
            \item \textit{Sottosistema Rotazionale}: Tecnica \textbf{Backstepping} integrata con azione integrale (PID).
        \end{enumerate}
        \item \textbf{Vantaggi}:
        \begin{itemize}
            \item Stabilità asintotica globale dimostrata (Lyapunov).
            \item Superiore reiezione dei disturbi rispetto ai PID standard.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- SEZIONE 4 ---
\section{Disaccoppiamento e Controllo Robusto Adattivo}

\begin{frame}{Disaccoppiamento e SMC [3]}
    Soluzione per la natura sottattuata dei velivoli ibridi (es. Coaxial Tilt-Rotor).
    \vspace{0.3cm}
    \begin{block}{Strategia a Due Anelli}
        \begin{itemize}
            \item \textbf{Decoupler}: Isola le dinamiche traslazionali da quelle rotazionali trasformando gli ingressi virtuali in forze reali e rollio desiderato.
            \item \textbf{Adaptive Backstepping SMC (ABSMC)}:
            \begin{itemize}
                \item \textit{Legge Adattiva}: Stima online il limite superiore dei disturbi ($\hat{\eta}_1$) per modulare il guadagno.
                \item \textit{Riduzione Chattering}: Uso della funzione di saturazione $\text{sat}(s)$ nello strato limite.
            \end{itemize}
        \end{itemize}
    \end{block}
\end{frame}

\section{Modellazione e Controllo Robusto per UAV Coaxial Tilt-Rotor}

\begin{frame}{Coaxial Tilt-Rotor (CTRUAV) [4]}
    Configurazione: 2 coppie di rotori coassiali anteriori + 1 rotore fisso di coda.
    \vspace{0.2cm}
    \begin{columns}
        \column{0.6\textwidth}
        \begin{itemize}
            \item \textbf{Caratteristiche}: Spinta elevata, annullamento naturale del momento giroscopico, dimensioni compatte.
            \item \textbf{Controllo Gerarchico}:
            \begin{itemize}
                \item Loop Esterno: Backstepping (Posizione).
                \item Loop Interno: Robust Adaptive ABSMC (Assetto).
            \end{itemize}
        \end{itemize}

% --- SEZIONE 6 (NUOVA, AGGIUNTA DAL TESTO) ---
\section{Controllo Sliding Mode "Smoothed" e Non-Overshooting}

\begin{frame}{SMC "Smoothed" e Continuo [5]}
    Una strategia recente per eliminare totalmente il \textit{chattering} e le sovraelongazioni (\textit{overshoot}).
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Innovazione}: Sostituzione della funzione $\text{sign}(s)$ o $\text{sat}(s)$ con la tangente iperbolica:
        \begin{equation}
             u(t) = -k \cdot \tanh(\rho \cdot s(t))
        \end{equation}
        \item \textbf{Vantaggi}:
        \begin{itemize}
            \item \textit{Continuità}: Segnale di controllo liscio e derivabile (meno stress sugli attuatori).
            \item \textit{Non-Overshooting}: Struttura progettata per convergere senza superare il target.
            \item \textit{Robustezza}: Per $\rho$ elevato, approssima le proprietà ideali dello SMC.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- SEZIONE 7 ---
\section{Strategie di Volo e Allocazione del Controllo}

\begin{frame}{Volo Autonomo "Full-Mode" [6]}
    Realizzazione pratica su Tricottero Tilt-Rotor senza modelli aerodinamici complessi.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Gestione Volo}: Macchina a stati (Rotor $\to$ Conversion $\to$ Fixed-wing $\to$ Reconversion).
        \item \textbf{Control Allocation (Mixer)}:
        \begin{itemize}
            \item Distribuisce i pesi tra rotori e superfici alari in base all'angolo di tilt.
            \item \textit{Imbardata in Hovering}: Gestita tramite \textbf{tilt differenziale} dei rotori anteriori.
        \end{itemize}
    \end{itemize}

\end{frame}

% --- SEZIONE 8 ---
\section{Controllo Sliding Mode a Tempo Fisso}

\begin{frame}{Fixed-Time Stability [7]}
    Garantire la convergenza entro un tempo limitato noto a priori, indipendentemente dallo stato iniziale.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Nonsingular Fixed-Time SMC (NFTSMC)}:
        \begin{itemize}
            \item \textit{Superficie Non Singolare}: Evita instabilità numeriche.
            \item \textit{Stima Adattiva}: Aggiornamento real-time per compensare i disturbi.
        \end{itemize}
        \item \textbf{Risultati}:
        \begin{itemize}
            \item Convergenza degli errori di assetto in $< 0.3\text{s}$.
            \item Elevata robustezza contro vento e variazioni di carico.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- SEZIONE 9 ---
\section{Allocazione del Controllo tramite Spinta Differenziale}

\begin{frame}{Thrust-Based Control (LTA) [8]}
    Applicazione di logiche UAV su dirigibili (Lighter-Than-Air) senza superfici aerodinamiche mobili.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Concetto}: Sostituzione di alettoni/timoni con la modulazione della spinta (\textit{Differential Thrust}).
        \item \textbf{Requisito}: Motori reversibili.
        \item \textbf{Mapping (Configurazione a X)}:
        \begin{itemize}
            \item \textit{Pitch}: Spinta differenziale tra rotori Superiori e Inferiori.
            \item \textit{Yaw}: Spinta differenziale tra rotori Destri e Sinistri.
            \item \textit{Roll}: Configurazione incrociata (es. Basso-DX + Alto-SX).
        \end{itemize}
    \end{itemize}
\end{frame}

% --- CONCLUSIONI ---
\section{Conclusioni}

\begin{frame}{Conclusioni}
    L'analisi dello stato dell'arte evidenzia:
    \begin{itemize}
        \item La necessità di gestire la complessità dei velivoli \textbf{ibridi} nella fase di transizione.
        \item L'evoluzione dei controllori verso soluzioni \textbf{robuste e adattive} (SMC, Backstepping, Fixed-Time) per mitigare disturbi e incertezze.
        \item L'importanza del \textbf{Control Allocation} per sfruttare la sovra-attuazione (Tilt-rotor) o compensare l'assenza di superfici aerodinamiche (Thrust-based).
    \end{itemize}
\end{frame}
