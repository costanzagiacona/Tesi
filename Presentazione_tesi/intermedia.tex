\section{Sistemi di Riferimento e Dinamica}

\begin{frame}{Sistemi di Riferimento}
    Per descrivere la dinamica si adottano due sistemi destrorsi:
    \begin{columns}
        \column{0.6\textwidth}
        \begin{itemize}
            \item \textbf{Sistema Inerziale (Earth Frame) $\mathcal{F}_g$}: 
            \begin{itemize}
                \item Fisso al terreno $(O_g, X_g, Y_g, Z_g)$.
                \item Posizione: $P_g = [x_g, y_g, z_g]^T$.
                \item Orientamento: $\Theta_g = [\phi, \theta, \psi]^T$.
            \end{itemize}
            \vspace{0.5cm}
            \item \textbf{Sistema Solidale (Body Frame) $\mathcal{F}_b$}:
            \begin{itemize}
                \item Origine nel CoM, assi $(X_b, Y_b, Z_b)$.
                \item Velocità lineari: $V_b = [u, v, w]^T$.
                \item Velocità angolari: $\Omega_b = [p, q, r]^T$.
            \end{itemize}
        \end{itemize}
        
        \column{0.4\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\textwidth]{sist_rif.png}
            \caption{Sistemi di riferimento.}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{Equazioni della Dinamica}
    Modello Newton-Eulero per corpo rigido a 6 gradi di libertà:
    
    \begin{block}{Equazioni Vettoriali}
    \begin{equation}
        \begin{cases}
            m \dot{V}_b + \Omega_b \times (m V_b) = F_{\mathrm{tot}} \\
            I_b \dot{\Omega}_b + \Omega_b \times (I_b \Omega_b) = M_{\mathrm{tot}}
        \end{cases}
    \end{equation}
    \end{block}
    
    Dove:
    \begin{itemize}
        \item $F_{\mathrm{tot}} = F_{\mathrm{th}} + F_g + F_{\mathrm{aero}}$
        \item $M_{\mathrm{tot}} = M_{\mathrm{th}} + M_{\mathrm{drag}} + M_{\mathrm{aero}} + M_{\mathrm{gyro}}$
    \end{itemize}
\end{frame}

\begin{frame}{Forza Totale e Componenti}
    La forza generata dai propulsori nel body frame è:
    {\footnotesize 
    \begin{equation}
        F_{\mathrm{th}} = 
        \begin{bmatrix}
        c_{\theta_1} f_1 + c_{\theta_2} f_2 + c_{\theta_3} c_{\theta_4} f_3 \\
        c_{\theta_3} s_{\theta_4} f_3 \\
        -s_{\theta_1} f_1 - s_{\theta_2} f_2 - s_{\theta_3} f_3
        \end{bmatrix}
    \end{equation}
    }
    
    A cui si aggiungono gravità e aerodinamica:
    \begin{equation}
        F_g = mg \begin{bmatrix} -\sin\theta \\ \cos\theta \sin\phi \\ \cos\theta \cos\phi \end{bmatrix}, \quad
        F_{\mathrm{aero}} \approx 
        \begin{bmatrix}
            -\frac{1}{2}\rho S C_D v_x |v_x| \\
            0 \\
            -\frac{1}{2}\rho S C_L v_x^2
        \end{bmatrix}
    \end{equation}
\end{frame}

\section{Architettura di Controllo}

\begin{frame}{Strategia di Controllo}
    È stata implementata un'architettura **gerarchica a cascata**.
    
    \begin{block}{Concetto Chiave}
        La spinta totale $T$ sostiene il drone, mentre la sua orientazione (tramite rollio e beccheggio) genera le forze planari.
    \end{block}

    \vspace{0.5cm}
    \textbf{Struttura degli Anelli:}
    \begin{enumerate}
        \item \textbf{Outer Loop (Quota):} Dinamica verticale lenta. Usa \textbf{SMC} per calcolare la spinta $F_{req,z}$.
        \item \textbf{Inner Loop (Posizione e Assetto):} 
        \begin{itemize}
            \item \textit{Posizione ($x,y$):} Genera angoli di riferimento $\phi_{des}, \theta_{des}$ (SMC).
            \item \textit{Assetto ($\phi, \theta, \psi$):} Insegue gli angoli e stabilizza le velocità angolari (PID).
        \end{itemize}
    \end{enumerate}
\end{frame}

\section{Algoritmi di Controllo}

\begin{frame}{Sliding Mode Control (SMC)}
    \textbf{Controllo Verticale:}
    Superficie di scorrimento $\sigma_z = \dot{e}_z + \lambda_z e_z$. Legge di controllo per $F_{req,z}$:
    \begin{equation}
        F_{req,z} = m g + F_{aero,z} - m \lambda_z \dot{e}_z - m K_z \tanh\left(\frac{\sigma_z}{\Phi}\right)
    \end{equation}
    
    \vspace{0.5cm}
    \textbf{Controllo Planare (Generazione Riferimenti):}
    Inversione della dinamica per calcolare l'assetto necessario:
    \begin{align}
        \phi_{des} &= \arcsin\left( \frac{m}{T} \cdot u_y \right) \quad (\text{Controllo laterale}) \\
        \theta_{des} &= \arcsin\left( - \frac{m}{T} \cdot u_x \right) \quad (\text{Controllo longitudinale})
    \end{align}
\end{frame}

\section{Control Allocation}

\begin{frame}{Mixer e Allocazione Dinamica}
    Il sistema non è risolvibile con una semplice matrice pseudoinversa a causa della non-linearità geometrica. Si usa un approccio sequenziale.
    
    \vspace{0.5em}
    \textbf{1. Equilibrio Longitudinale (Pitch):}
    Il motore di coda ($T_3$) bilancia il momento $M_y$ e parte della spinta verticale.
    \begin{equation}
        \omega_3^2 = \frac{F_{req} d_{mx} - M_y}{K \sin\theta_3 (d_{mx} - d_{tx}) + b \cos\theta_3}
    \end{equation}
    
    \textbf{2. Equilibrio Laterale (Roll):}
    La spinta residua necessaria viene divisa tra i motori anteriori.
    \begin{equation}
        T_1 = T_i - \frac{M_x}{2 d_{my}}, \quad T_2 = T_i + \frac{M_x}{2 d_{my}}
    \end{equation}
    
    \textbf{3. Imbardata (Yaw):}
    Si calcola il tilt necessario ($\delta$) del rotore di coda.
    \begin{equation}
        \delta_{des} = \text{sat}\left( \arcsin\left( \frac{M_z}{2 T_i d_{my}} \right), \delta_{max} \right)
    \end{equation}
\end{frame}
