\begin{frame}{Sistemi di Riferimento}
    \small 
    Per descrivere la dinamica del tricottero si adottano due sistemi destrorsi:
    
    \begin{itemize}
        \item \textbf{Sistema Inerziale ($\mathcal{F}_g$):} 
        Posizione $\bm{P}_g = [x, y, z]^T$, Angoli di assetto $\bm{\Theta}_g = [\phi, \theta, \psi]^T$.
        \item \textbf{Sistema Solidale ($\mathcal{F}_b$):}
        Origine nel Centro di Massa (CoM). Velocità lineari $\bm{V}_b = [u, v, w]^T$, Velocità angolari $\bm{\Omega}_b = [p, q, r]^T$.
    \end{itemize}

\begin{frame}{Cinematica: Trasformazioni tra sistemi}
    Le equazioni cinematiche legano le variazioni temporali delle coordinate inerziali alle velocità misurate nel sistema solidale al corpo:
    
    \begin{equation}
        \bm{\dot{P}}_{g} = \mathbf{R}_{b \to g}(\bm{\Theta}_g) \bm{V}_{b}
    \end{equation}
    \begin{equation}
        \bm{\dot{\Theta}}_{g} = \mathbf{J}_{b \to g}(\bm{\Theta}_g) \bm{\Omega}_{b}
    \end{equation}

    In forma compatta per lo stato cinematico:
    \begin{equation}
        \begin{bmatrix}
            \bm{\dot{P}}_g \\ 
            \bm{\dot{\Theta}}_g
        \end{bmatrix}
        =
        \begin{bmatrix}
            \mathbf{R}_{b \to g} & \mathbf{0} \\ 
            \mathbf{0} & \mathbf{J}_{b \to g}
        \end{bmatrix}
        \begin{bmatrix}
            \bm{V}_b \\
            \bm{\Omega}_b
        \end{bmatrix}
    \end{equation}
\end{frame}

\begin{frame}{Configurazione Geometrica e Attuazione}
    I rotori anteriori sono avanzati rispetto al CoM ($d_x > 0$), introducendo un accoppiamento intrinseco tra spinta e beccheggio.
    
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \textbf{Vettori Posizione (Bracci):}
            \small
            \begin{itemize}
                \item \textbf{Rotore 1 (DX):} $\bm{r}_1 = [d_{mx}, d_{my}, 0]^T$
                \item \textbf{Rotore 2 (SX):} $\bm{r}_2 = [d_{mx}, -d_{my}, 0]^T$
                \item \textbf{Rotore 3 (Coda):} $\bm{r}_3 = [-d_{tx}, 0, 0]^T$
            \end{itemize}
            
            \vspace{0.2cm}
            \footnotesize
            \textcolor{red}{\textbf{Implicazione:}} La spinta totale non è applicata nel CoM; il bilanciamento del Pitch richiede una distribuzione di potenza asimmetrica tra rotori anteriori e di coda.
        \end{column}

\end{frame}

\begin{frame}{Equazioni della Dinamica: Newton-Eulero}
    Modello a 6 gradi di libertà per corpo rigido soggetto a forze e momenti esterni:
    
    \begin{block}{Equazioni del Moto nel Body Frame}
    \begin{equation}
        \begin{cases}
            m \bm{\dot{V}}_b + \bm{\Omega}_b \times (m \bm{V}_b) = \bm{F}_{\text{tot}} \\
            \mathbf{I}_b \bm{\dot{\Omega}}_b + \bm{\Omega}_b \times (\mathbf{I}_b \bm{\Omega}_b) = \bm{M}_{\text{tot}}
        \end{cases}
    \end{equation}
    \end{block}
    
    Scomposizione dei contributi:
    \begin{itemize}
        \item $\bm{F}_{\text{tot}} = \bm{F}_{\text{th}} + \bm{F}_g + \bm{F}_{\text{aero}}$
        \item $\bm{M}_{\text{tot}} = \bm{M}_{\text{th}} + \bm{M}_{\text{tor}} + \bm{M}_{\text{aero}} + \bm{M}_{\text{gyro}} + \bm{M}_{\text{pinna}}$
    \end{itemize}
\end{frame}

\begin{frame}{Scomposizione della Forza Totale}
    La forza totale agente sul tricottero è la risultante di tre contributi principali:
    \begin{equation}
        \bm{F}_{\text{tot}} = \bm{F}_{\text{th}} + \bm{F}_g + \bm{F}_{\text{aero}}
    \end{equation}

    \begin{itemize}
        \item \textbf{Spinta ($\bm{F}_{\text{th}}$):} Generata dai tre rotori.
        \begin{equation}
            \bm{F}_{\text{th}} = 
            \begin{bmatrix}
                k\omega_1^2 \cos\theta_1 + k\omega_2^2 \cos\theta_2 + k\omega_3^2 \cos\theta_3 \cos\theta_4 \\
                k\omega_3^2 \cos\theta_3 \sin\theta_4 \\
                -k\omega_1^2 \sin\theta_1 - k\omega_2^2 \sin\theta_2 - k\omega_3^2 \sin\theta_3
            \end{bmatrix}
        \end{equation}
        \item \textbf{Gravità ($\bm{F}_g$):} Proiettata dal sistema inerziale al body frame tramite la matrice $R_{b \to g}^T$:
        \begin{equation}
            \bm{F}_g = \begin{bmatrix} -\sin\theta \\ \cos\theta \sin\phi \\ \cos\theta \cos\phi \end{bmatrix} mg
        \end{equation}
        \item \textbf{Aerodinamica ($\bm{F}_{\text{aero}}$):} Include Lift e Drag, dipendenti dal quadrato della velocità $v^2$ e dalla funzione $\text{sgn}(v)$ per la direzione.
    \end{itemize}
\end{frame}


\begin{frame}{Forze Aerodinamiche}
    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            \footnotesize
            La resistenza della fusoliera ($\bm{D}_{\text{body}}$) è modellata lungo gli assi principali del \textit{Body Frame}:
            \begin{equation}
                \bm{D}_{\text{body}} = -\frac{1}{2} \rho 
                \begin{bmatrix}
                    \text{sgn}(u) v_x^2 s_{body,x} C_{Dx} \\
                    \text{sgn}(v) v_y^2 s_{body,y} C_{Dy} \\
                    \text{sgn}(w) v_x^2 s_{body,z} C_{Dz}
                \end{bmatrix}
            \end{equation}
            
            Il contributo alare ($\bm{F}_{\text{wing}}$) in crociera è espresso in funzione della velocità longitudinale $u$:
            \begin{equation}
                \bm{F}_{\text{wing}} \approx 
                \begin{bmatrix}
                    -\frac{1}{2}\rho S C_D(v_x) v_x |v_x| \\
                    0 \\
                    -\frac{1}{2}\rho S C_L(v_x) v_x^2
                \end{bmatrix}
            \end{equation}
            
            \vspace{0.2cm}
            I coefficienti $C_L$ e $C_D$ sono funzioni non lineari dell'angolo d'attacco $\alpha = \arctan(v_z/v_x)$.
        \end{column}

        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\linewidth]{forze_x.jpg}
            \captionof{figure}{Convenzione delle forze aerodinamiche nel piano longitudinale.}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Momento di Thrust}
    \begin{equation}
        \bm{M}_{\text{tot}} = \bm{M}_{\text{th}} + \bm{M}_{\text{tor}} + \bm{M}_{\text{aero}} + \bm{M}_{\text{gyro}} + \bm{M}_{\text{pinna}}
    \end{equation}
    Il momento totale di thrust $\bm{M}_{\text{th}}$ è calcolato come somma dei momenti prodotti dai tre rotori rispetto al baricentro:
    
    \begin{equation*}
        \bm{M}_{\text{th}} = \sum_{i=1}^{3} (\bm{r}_i \times \bm{F}_{th_{i}})
    \end{equation*}

    Utilizzando i bracci forniti e le componenti della forza di spinta, otteniamo:

    \begin{equation}
        \bm{M}_{\text{th}} = 
        \begin{bmatrix} 
        -d_{my} (k\omega_1^2 \sin\theta_1) + d_{my} (k\omega_2^2 \sin\theta_2) \\ 
        d_{mx} (k\omega_1^2 \sin\theta_1) + d_{mx} (k\omega_2^2 \sin\theta_2) + d_{tx} (k\omega_3^2 \sin\theta_3) \\ 
        -d_{my} (k\omega_1^2 \cos\theta_1) + d_{my} (k\omega_2^2 \cos\theta_2) + d_{tx} (k\omega_3^2 \cos\theta_3 \sin\theta_4)
        \end{bmatrix}
    \end{equation}
\end{frame}

\begin{frame}{Momento Torcente dei Rotori}
    Il momento torcente totale è la somma vettoriale dei momenti generati da ciascun rotore, dovuti alla resistenza aerodinamica e all'inerzia del rotore stesso:
    
    \begin{equation*}
        \bm{M}_{\text{tor}} = \sum_{i=1}^{3} \bm{\tau}_i = \sum_{i=1}^{3} (b \omega_i^2 + I_{rot_{i,z}} \dot{\omega}_i) \hat{e}_{thrust,i}
    \end{equation*}

    Espandendo i termini secondo l'orientamento dei rotori ($\theta_i$):

    \begin{equation}
        \bm{M}_{\text{tor}} = 
        \begin{bmatrix} 
        \tau_{1_x} - \tau_{2_x} + \tau_{3_x} \\ 
        \tau_{3_y} \\ 
        -\tau_{1_z} + \tau_{2_z} - \tau_{3_z} 
        \end{bmatrix} =
        \begin{bmatrix} 
        (b\omega_1^2 + I_{r}\dot{\omega}_1) c_{\theta_1} - (b\omega_2^2 + I_{r}\dot{\omega}_2) c_{\theta_2} + (b\omega_3^2 + I_{r}\dot{\omega}_3) c_{\theta_3} c_{\theta_4} \\ 
        (b\omega_3^2 + I_{r}\dot{\omega}_3) \cos\theta_3 \sin\theta_4 \\ 
        -(b\omega_1^2 + I_{r}\dot{\omega}_1) s_{\theta_1} + (b\omega_2^2 + I_{r}\dot{\omega}_2) s_{\theta_2} - (b\omega_3^2 + I_{r}\dot{\omega}_3) \sin\theta_3
        \end{bmatrix}
    \end{equation}

    \begin{itemize}
        \item $b$: Coefficiente di resistenza aerodinamica dell'elica.
        \item $I_{rot_{i,z}}$: Momento d'inerzia del rotore rispetto all'asse di rotazione.
        \item $\hat{e}_{thrust,i}$: Versore dell'asse del rotore inclinato.
    \end{itemize}
\end{frame}

\begin{frame}{Momento Aerodinamico}
    Il momento aerodinamico totale $\bm{M}_{\text{aero}}$ rispetto al baricentro è il momento generati dalle ali:
    
    \begin{equation*}
        \bm{M}_{\text{aero}} = \sum_{i \in \{dx, sx\}} (\mathbf{r}_{ala_i} \times \mathbf{F}_{aero, wing})
    \end{equation*}

    \begin{equation}
        \bm{M}_{\text{aero}} = 
        \sum_{i \in \{dx, sx\}} 
        \begin{bmatrix} 
        r_{i_y} F_{z} \\ 
        r_{i_z} F_{x} - r_{i_x} F_{z} \\ 
        -r_{i_y} F_{x} 
        \end{bmatrix}
    \end{equation}
\end{frame}

\begin{frame}{Momento Giroscopico dovuto al tilting dei rotori}
    Il momento giroscopico nasce dalla variazione dell'asse di rotazione dei propulsori (tilting). È definito dal prodotto vettoriale tra la velocità di tilting e il momento angolare del rotore:

    \begin{equation}
        \bm{M}_{\text{giro}} = \sum_{i=1}^{3} \mathbf{J}_{r} (\bm{\omega}_{\text{tilt},i} \times \bm{\omega}_{r,i})
    \end{equation}

    % Il resizebox adatta automaticamente l'equazione alla larghezza del testo
    \begin{equation}
    \resizebox{0.95\textwidth}{!}{%
        $
        \mathbf{M}_{\text{giro}} = 
        \begin{bmatrix}
            -I_{\text{rot}_{12_z}} \omega_{\text{tilt}_1} \sin(\theta_1)\omega_1 
            + I_{\text{rot}_{12_z}} \omega_{\text{tilt}_2} \sin(\theta_2)\omega_2 
            - I_{\text{rot}_{3_z}} \omega_{\text{tilt}_{3,xz}} \sin(\theta_3)\omega_3 
            - I_{\text{rot}_{3_y}} \omega_{\text{tilt}_{3,xy}} \cos(\theta_3) \sin(\theta_4)\omega_3 
            \\[1em] % Spaziatura extra tra le righe
            + I_{\text{rot}_{3_x}} \omega_{\text{tilt}_{3,xy}} \cos(\theta_3) \cos(\theta_4)\omega_3 
            \\[1em]
            -I_{\text{rot}_{12_x}} \omega_{\text{tilt}_1} \cos(\theta_1)\omega_1 
            + I_{\text{rot}_{12_x}} \omega_{\text{tilt}_2} \cos(\theta_2)\omega_2 
            - I_{\text{rot}_{3_x}} \omega_{\text{tilt}_{3,xz}} \cos(\theta_3) \cos(\theta_4)\omega_3
        \end{bmatrix}
        $
    }
    \end{equation}

    \vspace{0.2cm}
    \footnotesize
    Dove:
    \begin{itemize}
        \item $I_{\text{rot}_{12}}$ e $I_{\text{rot}_{3}}$: Momenti d'inerzia dei rotori anteriori e posteriore.
        \item $\omega_{\text{tilt}}$: Velocità angolari di attuazione dei servomotori.
        \item $\omega_{1,2,3}$: Velocità angolari delle eliche.
    \end{itemize}
\end{frame}

\begin{frame}{Momento Giroscopico dovuto al tricottero}
    Il momento giroscopico $\bm{M}_{\text{body}}$ rappresenta le coppie derivanti dalla rotazione del velivolo nel sistema di riferimento solidale (\textit{Body Frame}):
    
    \begin{equation*}
        \bm{M}_{\text{body}} = \bm{\Omega}_{B} \times (I_{B} \bm{\Omega}_{B})
    \end{equation*}

    Espandendo il prodotto vettoriale con la velocità angolare $\bm{\Omega}_{B} = [p, q, r]^\top$ e la matrice d'inerzia del tricottero $I_{B}$:

    \begin{equation}
        \bm{M}_{\text{body}} = 
        \begin{bmatrix} p \\ q \\ r \end{bmatrix} \times 
        \begin{bmatrix} I_{xx} & 0 & 0 \\ 0 & I_{yy} & 0 \\ 0 & 0 & I_{zz} \end{bmatrix}
        \begin{bmatrix} p \\ q \\ r \end{bmatrix} =
        \begin{bmatrix} 
        (I_{zz} - I_{yy}) qr \\ 
        (I_{xx} - I_{zz}) pr \\ 
        (I_{yy} - I_{xx}) pq 
        \end{bmatrix}
    \end{equation}

    \begin{itemize}
        \item Queste componenti rappresentano le coppie giroscopiche dovute alla distribuzione della massa del velivolo.
    \end{itemize}
\end{frame}

\begin{frame}{Momento della Pinna in Coda}
    Il momento generato dalla pinna di coda $\bm{M}_{pinna}$ agisce come contributo stabilizzante e smorzante, dipendente dalle velocità angolari e dalla velocità longitudinale:

    \begin{equation}
        \bm{M}_{pinna} = 
        \begin{bmatrix} M_{pinna,x} \\ M_{pinna,y} \\ M_{pinna,z} \end{bmatrix} =
        \begin{bmatrix} 
        -p \, \alpha_{0x} - \alpha_{1x} \, v_y^2 \\ 
        0 \\ 
        -r \, (\alpha_{0z} + \alpha_{1z} \, v_y^2) 
        \end{bmatrix}
    \end{equation}

    \begin{itemize}
        \item $p, r$: Velocità angolari rispettivamente di rollio ($X_B$) e imbardata ($Z_B$).
        \item $v_y$: Velocità longitudinale lungo l'asse $Y_B$.
        \item $\alpha_{0x}, \alpha_{1x}, \alpha_{0z}, \alpha_{1z}$: Coefficienti aerodinamici specifici della pinna.
    \end{itemize}

    \begin{block}{Osservazioni}
        La pinna non genera momento di beccheggio ($M_{pinna,y} = 0$) e i segni negativi indicano che il momento si oppone al moto rotatorio, garantendo stabilità direzionale.
    \end{block}
\end{frame}

\section{Sistema Dinamico}

\begin{frame}
    \vfill
    \centering
    {\Huge \textbf{Sistema Dinamico}}
    \vfill
\end{frame}

% Slide 1: Definizione del Problema
\begin{frame}{Definizione del Sistema Dinamico}
    Il modello matematico del tricottero è tradotto in un sistema di equazioni differenziali ordinarie (ODE) che descrivono l'evoluzione temporale dello stato.
    
    \vspace{1em}
    
    Le equazioni di stato sono formulate come:
    \begin{equation}
        \dot{x}(t) = f(x(t), u(t), t)
    \end{equation}
    
    \begin{itemize}
        \item $x(t) \in \mathbb{R}^{26}$: Vettore di stato.
        \item $u(t) \in \mathbb{R}^{7}$: Vettore degli ingressi di controllo.
        \item $f(\cdot)$: Dinamica non lineare (Newton-Eulero + Dinamica Attuatori).
    \end{itemize}
\end{frame}

% Slide 2: Variabili di Stato
\begin{frame}{Variabili di Stato}
    La parte relativa al corpo rigido comprende:
        \vspace{0.2cm}
        \begin{columns}[T]
            \begin{column}{0.48\textwidth}
                \textbf{Posizione e Velocità Lineare}
                \begin{itemize}
                    \item $x_{1,2,3} = [p_x, p_y, p_z]^T$: Posizione (Frame Inerziale).
                    \item $x_{4,5,6} = [v_{bx}, v_{by}, v_{bz}]^T$: Velocità (Body Frame).
                \end{itemize}
            \end{column}
            
            \begin{column}{0.48\textwidth}
                \textbf{Assetto e Velocità Angolari}
                \begin{itemize}
                    \item $x_{7,8,9} = [\phi, \theta, \psi]^T$: Angoli di Eulero (RPY).
                    \item $x_{10,11,12} = [p, q, r]^T$: Velocità angolari (Body Frame).
                \end{itemize}
            \end{column}

        \item \textbf{Vettorizzazione ($u_4, u_5, u_6$):} Angoli di tilt desiderati intorno a $Y_B$.

% Slide 5: Cinematica
\begin{frame}{Cinematica: Lineare e Rotazionale}
    \textbf{Velocità Lineari (Inerziali vs Body):}
    \begin{equation}
        \dot{p}_{inertial} = R_{b\rightarrow g}(\phi, \theta, \psi) \cdot v_{body}
    \end{equation}
    
    \textbf{Velocità Angolari (Eulero vs Body Rates):}
    La relazione è definita dalla matrice cinematica $T(\phi, \theta)$:
    \begin{equation}
        \begin{bmatrix} \dot{\phi} \\ \dot{\theta} \\ \dot{\psi} \end{bmatrix} = 
        \begin{bmatrix}
        1 & s_\phi t_\theta & c_\phi t_\theta \\
        0 & c_\phi & -s_\phi \\
        0 & \frac{s_\phi}{c_\theta} & \frac{c_\phi}{c_\theta}
        \end{bmatrix}
        \begin{bmatrix} p \\ q \\ r \end{bmatrix}
    \end{equation}
    \footnotesize{\textit{Singolarità per $\theta = \pm 90^\circ$ (Gimbal Lock).}}
\end{frame}

% Slide 6: Dinamica
\begin{frame}{Dinamica: Traslazionale e Rotazionale}
    L'accelerazione lineare nel body frame deriva dal bilancio delle forze:
    \begin{equation}
        \begin{bmatrix} \dot{u} \\ \dot{v} \\ \dot{w} \end{bmatrix} = \frac{1}{m} F_{TOT} - (\omega_{body} \times v_{body})
    \end{equation}
    
    L'evoluzione delle velocità angolari dipende dal momento d'inerzia $I_B$:  
    \begin{equation}
        \dot{\omega}_{body} = I_B^{-1} \left( M_{TOT} - \omega_{body} \times (I_B \omega_{body}) \right)
    \end{equation}

    I motori e i servi sono sistemi del \textbf{secondo ordine}. Esempio per il tilt ($\theta_i$):  
    \begin{align}
        \dot{x}_{2i-1} &= x_{2i} \quad \text{(Velocità)} \\
        \dot{x}_{2i} &= \underbrace{-2\zeta \omega_n x_{2i}}_{\text{Smorzamento}} + \underbrace{\omega_n^2 (u_{cmd} - x_{2i-1})}_{\text{Rigidezza/Errore}}
    \end{align}
\end{frame}

\begin{frame}{Configurazione fase di decollo}
    La prima fase di volo è quella di decollo verticale, durante la quale i motori sono posizionati verticalmente per generare una propulsione in grado di far sollevare il drone.
    
    \vspace{0.3cm}
    Si assume che in tale fase gli angoli di base dei rotori siano i seguenti:
    \begin{itemize}
        \item $\theta_1 = \theta_2 = \pi/2$
        \item $\theta_4 = -\pi/2$
        \item $\theta_3 = \theta_{3,eq}$: tale angolo permette di contrastare il momento torcente attorno all'asse $z$ causato dalla coppia di reazione del motore posteriore, non essendo presente un secondo rotore controrotante per il bilanciamento.
    \end{itemize}
\end{frame}

\begin{frame}{Calcolo dell'Angolo di Equilibrio $\theta_{3_{eq}}$}
    \small
    In configurazione di \textbf{Hovering}, è necessario calcolare un angolo di tilt $\theta_3$ tale da annullare il momento di imbardata residuo generato dalla reazione del rotore di coda.

    \vspace{0.2cm}
    \textbf{1. Bilancio dei Momenti su Z ($M_{z}^{tot} = 0$):}
    \footnotesize
    \begin{align*}
        M_{th,z} &= \underbrace{d_{my} k (\omega_{2}^2 c_{\theta_2} - \omega_{1}^2 c_{\theta_1})}_{\approx 0 \text{ per } \theta_{1,2} \approx 90^\circ} 
                  - d_{ty} k \omega_{3}^2 c_{\theta_3} c_{\theta_4} 
                  + d_{tx} k \omega_{3}^2 c_{\theta_3} s_{\theta_4} \\
        M_{tor,z} &= \underbrace{-b \omega_{1}^2 s_{\theta_1} + b \omega_{2}^2 s_{\theta_2}}_{\approx 0 \text{ (Simmetria } \omega_1=\omega_2 \text{)}} 
                   - b \omega_{3}^2 s_{\theta_3}
    \end{align*}
    
    \vspace{0.1cm}
    \normalsize
    \textbf{2. Semplificazione e Soluzione:}
    Considerando $\theta_4$ fisso (orientamento laterale) e imponendo l'equilibrio:
    \begin{equation*}
        \omega_3^2 \left( d_{tx} k \cos\theta_3 \sin\theta_4 - b \sin\theta_3 \right) = 0
    \end{equation*}
    Si ottiene l'angolo di equilibrio statico:  
    \begin{equation}
        \boxed{\theta_{3_{eq}} = \arctan\left( -\frac{k d_{tx}}{b}\right)}
    \end{equation}
    
\end{frame}

\begin{frame}{Architettura Gerarchica a Cascata}
    Sulla base dell'analisi dinamica, è stata implementata un'architettura di controllo a cascata che disaccoppia la dinamica lenta (posizione) da quella veloce (assetto).

    \begin{block}{Suddivisione Logica}
    \begin{enumerate}
        \item \textbf{Outer Loop (Posizione $x, y, z$):}
        Utilizza la tecnica \textbf{Sliding Mode Control (SMC)}.
        \begin{itemize}
            \item Calcola le forze richieste ($F_{req}$) per annullare l'errore di posizione.
            \item Effettua un \textit{mapping cinematico} traducendo le forze planari in angoli di assetto desiderati ($\phi_{des}, \theta_{des}$).
        \end{itemize}
        
        \item \textbf{Inner Loop (Assetto $\phi, \theta, \psi$):}
        Utilizza controllori \textbf{PID} per inseguire gli angoli desiderati calcolati dal loop esterno e stabilizzare le velocità angolari.
    \end{enumerate}
    \end{block}
    
    \vspace{0.2cm}
    \textbf{Peculiarità:} Il controllo di quota e quello planare sono \textbf{accoppiati}. La spinta totale $T$ (loop $z$) determina l'inclinazione del vettore spinta.
\end{frame}

\begin{frame}{Analisi di Stabilità (Metodo di Lyapunov)}
    
    Per garantire la stabilità asintotica globale della \textit{sliding surface} ($\sigma \to 0$), si utilizza il criterio di Lyapunov.

    \vspace{0.3cm}
    
    \begin{columns}[T]
        % --- COLONNA SINISTRA: I Passaggi Logici ---
        \begin{column}{0.6\textwidth}
            \textbf{1. Funzione Candidata:}
            \begin{equation*}
                V(\sigma) = \frac{1}{2}\sigma^2 > 0
            \end{equation*}
            
            \textbf{2. Derivata Temporale:}
            Sostituendo la dinamica a ciclo chiuso con disturbo $\Delta$:
            \begin{align*}
                \dot{V} &= \sigma \dot{\sigma} \\
                        &= -k \sigma \text{sgn}(\sigma) + \sigma \Delta \\
                        &\le -k |\sigma| + |\sigma| \delta_{\max}
            \end{align*}
        \end{column}
        
        % --- COLONNA DESTRA: La Condizione ---
        \begin{column}{0.4\textwidth}
            \centering
            \vspace{1cm}
            \begin{alertblock}{Condizione di Stabilità}
                Affinché la funzione decresca ($\dot{V} < 0$), il guadagno deve "battere" il disturbo:
                \vspace{0.2cm}
                \centering
                \large
                \boldmath
                $k > \delta_{\max}$
                \unboldmath
            \end{alertblock}
            
            \vspace{0.3cm}
            \small \itshape
            Il sistema è robusto se il guadagno di switching supera il limite superiore dell'incertezza.
        \end{column}
    \end{columns}

    % Grafico concettuale opzionale in basso
    \vspace{0.5cm}
    \centering
    \begin{tikzpicture}[scale=0.6]
        \draw[->] (0,0) -- (6,0) node[right] {$t$};
        \draw[->] (0,0) -- (0,2.5) node[above] {$V(\sigma)$};
        \draw[thick, blue] plot [domain=0:5.5, samples=100] (\x, {2*exp(-\x)});
        \node[blue] at (2, 1.5) {$\dot{V} < 0$};
        \draw[dashed] (0,0) node[left] {0} -- (6,0);
    \end{tikzpicture}

\end{frame}

\begin{frame}{Gestione dei Disturbi e Tuning del Guadagno}
    
    Il vantaggio principale dello SMC è che non richiede la conoscenza puntuale del disturbo $d(t)$, ma solo del suo upper bound.
    
    \begin{columns}[c]
        \begin{column}{0.5\textwidth}
            \textbf{Condizione di Robustezza:}
            $$ k > |\Delta(t)|_{\max} $$
            
            \vspace{0.2cm}
            \small
            \begin{itemize}
                \item \textcolor{green!60!black}{\textbf{Guadagno Ottimale:}} $k$ leggermente superiore al disturbo massimo atteso. Stabilità garantita e chattering ridotto.

\begin{frame}{Controllo Planare e Inversione Dinamica}
    Essendo il sistema \textbf{sotto-attuato} sugli assi $x, y$, si utilizza la spinta totale $T$ (calcolata dal loop $z$) per generare le componenti orizzontali tramite il rollio ($\phi$) e il beccheggio ($\theta$).
    
    \vspace{0.2cm}
    Gli SMC planari calcolano le forze virtuali ($F_{y,req}, F_{x,req}$), che vengono convertite in angoli di riferimento invertendo la dinamica:
    
    \begin{columns}[t]
        \column{0.48\textwidth}
        \centering
        \textbf{Generazione Rollio ($\phi_{des}$)}
        \footnotesize
        \begin{equation}
            \phi_{des} = \arcsin\left( \frac{F_{y,req}}{T} \right)
        \end{equation}
        \raggedright
        Compensa l'errore laterale $e_y$.

        \column{0.48\textwidth}
        \centering
        \textbf{Generazione Beccheggio ($\theta_{des}$)}
        \footnotesize
        \begin{equation}
            \theta_{des} = \arcsin\left( - \frac{F_{x,req}}{T} \right)
        \end{equation}
        \raggedright
        Compensa l'errore longitudinale $e_x$.
    \end{columns}
    
    \vspace{0.3cm}
    \begin{alertblock}{Ipotesi di Validità}
    L'inversione assume \textbf{piccoli angoli} ($\sin\theta \approx \theta$) e una \textbf{separazione delle scale temporali}: la dinamica di assetto (inner loop) deve essere sufficientemente più veloce di quella di posizione (outer loop).
    \end{alertblock}
\end{frame}

\begin{frame}{Inner Loop: Controllo di Assetto}
Per inseguire gli angoli di riferimento ($\phi_{des}, \theta_{des}, \psi_{des}$) è necessario calcolare dei momenti torcenti rispettivamente lungo gli assi $x, y, z$.
Tali quantità sono calcolate come uscite di tre PID.
\begin{itemize}
    \item \textbf{Roll:} è stato applicato un controllo Proporzionale-Derivativo
    \begin{equation}
        M_{x,req} = K_P (\phi_{des} - \phi) + K_D (\dot{\phi}_{des} - p)
    \end{equation}
    \item \textbf{Pitch:} è stato applicato un controllo Proporzionale-Integrale-Derivativo. Il termine integrale reagisce alla condizione in cui il centro di pressione non coincida perfettamente con il centro di massa.
    \begin{equation}
        M_{y,req} = K_P (\theta_{des} - \theta) + K_I \int (\theta_{des} - \theta) \,dt\ + K_D (\dot{\theta}_{des} - q)
    \end{equation}
    \item \textbf{Yaw:} è stato applicato un controllo Proporzionale-Derivativo
    \begin{equation}
        M_{z,req} = K_P (\psi_{des} - \psi) + K_D (\dot{\psi}_{des} - r)
    \end{equation}
\end{itemize}
\end{frame}

\begin{frame}{MMA: Quota e Pitch}
    Il controllo dell'equilibrio longitudinale richiede la risoluzione di un sistema accoppiato: ogni variazione di spinta per la quota ($F_{z,req}$) induce un momento di beccheggio, e viceversa.
    
    \vspace{0.2cm}
    Tramite il mixer si determina la ripartizione ottima della spinta tra i rotori anteriori e il rotore posteriore:
    \begin{equation}
        \begin{cases}
        T_{front} + k \omega_{3}^2 \sin(\theta_{3}) = F_{z,req} \\
        T_{front} d_{mx} - k \omega_{3}^2 \sin(\theta_{3}) d_{tx} + b \omega_{3}^2 \cos(\theta_{3}) \sin(\theta_{4}) = M_y
        \end{cases}
    \end{equation}
    
    \vspace{0.2cm}
    Esprimendo $T_{front}$ come $2 k \omega_{i}^2$ (con $i=1,2$):
    \begin{equation}
        \begin{cases}
        2 k \omega_{i}^2 + k \omega_{3}^2 \sin(\theta_3) = F_{z,req} \\
        2 k \omega_{i}^2 d_{mx} - k \omega_{3}^2 \sin(\theta_3) d_{tx} + b \omega_{3}^2 \cos(\theta_{3}) \sin(\theta_{4}) = M_{y}
        \end{cases}
    \end{equation}
\end{frame}

\begin{frame}{MMA: Quota e Pitch}
    Risolvendo il sistema per l'incognita $\omega_{3}^2$, si ricava la velocità angolare del rotore di coda:
    \begin{equation}
        \omega_3^2 = \frac{M_y - F_{z,req} d_{mx}}{k \sin(\theta_3) (d_{tx} - d_{mx}) + b \cos(\theta_3) \sin(\theta_4)}
    \end{equation}
    
    \vspace{0.3cm}
    Una volta determinato $T_3 = k \omega_3^2$, si calcola la spinta necessaria per il gruppo propulsivo anteriore ($T_{front}$) per garantire il sostentamento e l'assetto richiesto:
    \begin{equation}
        T_{front} = F_{z,req} - T_3 \sin(\theta_3)
    \end{equation}
    
    \vspace{0.3cm}
\end{frame}

\begin{frame}{MMA: Roll}
    \textbf{Equilibrio Laterale (Roll):}
    Il rollio viene raggiunto attraverso la spinta differenziale dei rotori anteriori.\\
    La spinta totale anteriore $T_{front}$ viene ripartita tra il motore sinistro (2) e destro (1) per generare il momento di rollio richiesto ($M_x$).
    Il sistema da risolvere è:
    \begin{equation}
        \begin{cases}
        T_1 + T_2 = T_{front} \\
        (T_2 - T_1) d_{my} = M_x
        \end{cases}
    \end{equation}
    Definendo la spinta media anteriore come $T_i = T_{front} / 2$, le spinte dei singoli motori si ottengono per somma e differenza rispetto al termine differenziale legato al momento:

    \begin{equation}
        T_1 = T_i - \frac{M_x}{2 d_{my}}, \quad T_2 = T_i + \frac{M_x}{2 d_{my}}
    \end{equation}
    Le rispettive velocità angolari si ricavano infine come $\omega_{1,2} = \sqrt{T_{1,2}/k}$.
\end{frame}

\begin{frame}{MMA: Yaw}
    \textbf{Controllo Imbardata (Yaw):}
    L'imbardata è controllata attraverso il tilt differenziale dei rotori anteriori ($\delta$).
    La relazione che lega il momento torcente $M_z$ all'angolo di inclinazione è approssimabile come:
    \begin{equation}
        M_z = T_{front} d_{my} \sin(\delta)
    \end{equation}
    
    \vspace{0.2cm}
    Invertendo la relazione, si calcola l'angolo di tilt desiderato:
    \begin{equation}
        \delta_{des} = \arcsin\left( \frac{M_{z,req}}{T_{front} d_{my}} \right)
    \end{equation}
    Gli angoli dei servomotori sono:
    \begin{equation}
        \begin{cases}
            \theta_1 = \pi/2 - \delta_{des}\\
            \theta_2 = \pi/2 + \delta_{des}
        \end{cases}
    \end{equation}
\end{frame}

\section{Controllo Volo Orizzontale}

\begin{frame}
    \vfill
    \centering
    {\Huge \textbf{Controllo Orizzontale}}
    \vfill
\end{frame}

\begin{frame}{Configurazione fase di crociera}
    Durante la fase di crociera i rotori anteriori sono posizionati orizzontalmente per generare una spinta in avanti in grado di far avanzare il drone,
    mentre il rotore posteriore è spento per evitare instabilità generate dal momento torcente lungo l'asse $x$.
    
    \vspace{0.3cm}
    Si assume che in tale fase gli angoli di base dei rotori siano i seguenti:
    \begin{itemize}
        \item $\theta_1 = \theta_2 = 0$
        \item $\theta_4$ e $\theta_3$ sono irrilevanti.
    \end{itemize}
    In tale configurazione, il drone perde gradi di libertà e il controllo è affidato unicamente a spinta e tilting dei rotori anteriori.
    \vspace{0.2cm}
    \\La difficoltà è incrementata dall'assenza di superfici a geometria variabile, le quali vengono tipicamente usate per il controllo di assetto di velivoli \textit{fixed wing}.
\end{frame}

\begin{frame}{Architettura di Controllo: Struttura a Cascata}
    Il sistema di controllo per la fase di volo orizzontale (\textit{Fixed-Wing Mode}) adotta un'architettura a \textbf{loop in cascata} per disaccoppiare la dinamica lenta (traslazionale) da quella veloce (rotazionale).

    \begin{block}{Separazione delle Dinamiche}
        \begin{itemize}
            \item \textbf{Outer Loop (Guidance):} Opera a bassa frequenza. Calcola le forze richieste ($F_{x,req}, F_{z,req}$) per inseguire le traiettorie di posizione e velocità e generare un angolo di riferimento ($\phi_{des}$) per compensare l'errore lungo $y$.
            \item \textbf{Inner Loop (Attitude):} Opera ad alta frequenza. Stabilizza l'assetto ($\phi, \theta, \psi$) e gestisce le velocità angolari.
            \item \textbf{Motor Mixing Algorithm:} Mappa le forze e i momenti richiesti sui comandi degli attuatori (RPM motori e angoli servi).
        \end{itemize}
    \end{block}
    
\end{frame}

\begin{frame}{Outer Loop: Controllo di Posizione}
    Il controllore esterno utilizza un approccio PID con \textbf{Feedforward Aerodinamico} per compensare le non linearità note.
    
    \vspace{0.2cm}
    \textbf{Controllo di Quota ($z$)} \\
    Il primo obiettivo di controllo è il mantenimento della quota raggiunta in hovering.\\
    Si definisce l'errore $e_z = z_{des} - z$ e si calcola l'accelerazione necessaria per contrastare la gravità attraverso un PID:
    \begin{equation}
        a_{z,cmd} = K_P e_z + K_I \int e_z dt + K_D \dot{e}_z
    \end{equation} 
    La forza richiesta lungo z è pari a:
    \begin{equation}
        F_{z,req} = m (g - a_{z,cmd}) - L_{wing}(v)
        \label{eq:fz}
    \end{equation}
    dove $L_{wing} = \frac{1}{2}\rho S C_L v^2$ è il termine di feedforward della portanza.\\
\end{frame}

\begin{frame}{Outer Loop: Controllo di Velocità}
    L'obiettivo è raggiungere la velocità di crociera $v_{x,des} = 25 m/s$.\\
    \vspace{0.2cm}
    \textbf{Controllo di Velocità ($v_x$)} \\
    Definito l'errore $e_{v} = v_{x,des} - v_x$, la forza propulsiva richiesta è ottenuta tramite un controllore Proporzionale-Integrale:
    \begin{equation}
        F_{x,req} = D_{x}(v) + \underbrace{K_P e_v + K_I \int e_v dt + K_D \dot{e}_v}_{\text{Azione PI}}
        \label{eq:fx}
    \end{equation}
    dove $D_{x}$ rappresenta la stima della resistenza aerodinamica lungo l'asse $x$.\\
    Anche in questo caso si compensa la non linearità tramite un termine di feedforward.
\end{frame}



% ============================================================
% INNER LOOP
% ============================================================
\begin{frame}{Inner Loop: Controllo d'Assetto}
    
    In configurazione di crociera, i rotori generano spinta propulsiva (asse $X_B$). Il controllo dei momenti avviene tramite \textbf{Thrust Vectoring}, emulando le superfici aerodinamiche classiche.

    \begin{columns}[T]
        \column{0.33\textwidth}
        \centering
        \textbf{Pitch ($\theta$)} \\
        \textit{"Elevator"}
        \begin{itemize}
            \item \small{Azione: Tilt Collettivo}
            \item \small{Eq: $\theta_1 = \theta_2$}
            \item \small{Effetto: Modula la componente $F_z$ globale per alzare/abbassare il muso.}
        \end{itemize}

        \column{0.33\textwidth}
        \centering
        \textbf{Roll ($\phi$)} \\
        \textit{"Ailerons"}
        \begin{itemize}
            \item \small{Azione: Tilt Differenziale}
            \item \small{Eq: $\theta_1 = -\theta_2$}
            \item \small{Effetto: Genera una coppia di forze verticali opposte.}
        \end{itemize}

        \column{0.33\textwidth}
        \centering
        \textbf{Yaw ($\psi$)} \\
        \textit{"Rudder"}
        \begin{itemize}
            \item \small{Azione: Spinta Differenziale}
            \item \small{Eq: $T_1 \neq T_2$}
            \item \small{Effetto: Genera una coppia di trazione asimmetrica.}
        \end{itemize}
    \end{columns}
\end{frame}

% MIXER 
% ============================================================
\begin{frame}{MMA 1: Sintesi del Vettore Spinta}
    Il primo compito del Mixer è combinare le richieste di forza cartesiane ($F_{x,req}, F_{z,req}$) provenienti dall'Outer Loop in un unico vettore di spinta intensiva e direzionale.

    \vspace{0.2cm}
    \textbf{1. Magnitudo della Spinta Totale}
    La spinta totale necessaria per soddisfare le richieste di accelerazione è data dalla norma euclidea:
    \begin{equation}
        T_{tot} = \sqrt{F_{x,req}^2 + F_{z,req}^2} \quad [N]
    \end{equation}

    \vspace{0.2cm}
    \textbf{2. Orientamento nel Frame Inerziale}
    L'angolo di inclinazione del vettore spinta rispetto al sistema inerziale è:
    \begin{equation}
        \alpha_{inertial} = \arctan2\left( F_{z,req}, F_{x,req} \right)
    \end{equation}    
\end{frame}

% --- SLIDE 2: Trasformazione di Coordinate ---
\begin{frame}{MMA 2: Trasformazione Inerziale $\to$ Body}
    I servomotori sono solidali alla fusoliera del drone. Pertanto, l'angolo comandato ai servi ($\theta_1, \theta_2$) deve compensare l'assetto del velivolo (Pitch $\theta$).

    \begin{block}{Disaccoppiamento Geometrico}
        Affinché il vettore spinta rimanga orientato correttamente nello spazio inerziale indipendentemente dall'oscillazione del drone, si applica la trasformazione:
        \begin{equation}
            \alpha_{base} = \alpha_{inertial} - \theta_{body}
        \end{equation}
    \end{block}

    \vspace{0.3cm}
    \textbf{Interpretazione Fisica:}
    \begin{itemize}
        \item Se il drone cabra ($\theta > 0$), i motori devono ruotare verso il basso (diminuire $\alpha$) per mantenere la spinta orizzontale.
        \item Questo passaggio rende la dinamica di traslazione \textbf{invariante} rispetto alle perturbazioni di assetto (entro i limiti di saturazione dei servi).
    \end{itemize}
\end{frame}

% --- SLIDE 3: Distribuzione e Mixing ---
\begin{frame}{MMA 3: Attuatori}
    Infine, si sovrappongono i comandi di stabilizzazione ($u_\phi, u_\theta, u_\psi$) generati dall'Inner Loop al vettore base calcolato.

    \vspace{0.2cm}
    \textbf{A. Comandi ai Motori (Spinta Differenziale)}
    L'imbardata ($Yaw$) è controllata tramite la differenza di spinta tra i due rotori:
    \begin{equation}
        \begin{cases}
            T_{1} = \frac{T_{tot}}{2} - u_{\psi} \\
            T_{2}  = \frac{T_{tot}}{2} + u_{\psi}
        \end{cases}
        \implies \omega_{i} = \sqrt{\frac{T_{i}}{K}}
    \end{equation}

    \vspace{0.2cm}
    \textbf{B. Comandi ai Servi (Tilt Differenziale e Collettivo)}
    Il rollio ($Roll$) è controllato differenzialmente, mentre il beccheggio ($Pitch$) si somma collettivamente all'angolo base:
    \begin{equation}
        \begin{cases}
            \theta_{1} = \alpha_{base} + u_{\theta} - u_{\phi} \\
            \theta_{2} = \alpha_{base} + u_{\theta} + u_{\phi}
        \end{cases}
    \end{equation}
    
\end{frame}

    \end{block}

% Analisi Critica
\begin{frame}{Analisi Comparativa e Critica}
    \begin{itemize}
        \item \textbf{Assetto}: In entrambe le fasi, l'errore sugli angoli di Eulero è trascurabile (ordine $10^{-3}$ o inferiore). Il sistema di stabilizzazione angolare è robusto.
        \item \textbf{Discrepanza in Crociera}: 
        \begin{itemize}
            \item L'RMSE di $u \approx 10$ suggerisce che il drone non raggiunge la velocità di crociera desiderata (o rimane fermo mentre il setpoint avanza).
            \item L'errore $P_e$ elevato indica una deriva laterale non compensata.
        \end{itemize}
        \item \textbf{Verticalità}: L'errore sulla quota ($P_d$) rimane eccellente in entrambi i casi, indicando un ottimo controllo dell'altitudine.
    \end{itemize}
\end{frame}

% Conclusioni
\begin{frame}{Conclusioni e Prospettive}
    \begin{alertblock}{Sintesi}
        Il sistema è estremamente preciso in hovering, ma fallisce nel mantenere le prestazioni durante la crociera.
    \end{alertblock}
    \textbf{Azioni Correttive Consigliate:}
    \begin{enumerate}
        \item Verificare i guadagni del controllore di velocità longitudinale ($u$).
        \item Investigare la causa della deriva sulla coordinata Est ($P_e$) in fase di traslazione.
        \item Implementare un'azione integrale più aggressiva per eliminare l'errore a regime in crociera.
    \end{enumerate}
\end{frame}
