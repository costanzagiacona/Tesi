\begin{frame}{Cinematica: Trasformazioni tra sistemi}
    Le equazioni cinematiche legano le variazioni temporali delle coordinate inerziali alle velocità misurate nel sistema body:
    
    \begin{equation}
        \bm{\dot{P}}_{g} = \mathbf{R}_{b \to g}(\bm{\Theta}_g) \bm{V}_{b}
    \end{equation}
    \begin{equation}
        \bm{\dot{\Theta}}_{g} = \mathbf{J}_{b \to g}(\bm{\Theta}_g) \bm{\Omega}_{b}
    \end{equation}

    In forma compatta per lo stato cinematico:
    \begin{equation}
        \begin{bmatrix}
            \bm{\dot{P}}_g \\ 
            \bm{\dot{\Theta}}_g
        \end{bmatrix}
        =
        \begin{bmatrix}
            \mathbf{R}_{b \to g} & \mathbf{0}_{3 \times 3} \\ 
            \mathbf{0}_{3 \times 3} & \mathbf{J}_{b \to g}
        \end{bmatrix}
        \begin{bmatrix}
            \bm{V}_b \\
            \bm{\Omega}_b
        \end{bmatrix}
    \end{equation}
\end{frame}

\begin{frame}{Configurazione Geometrica e Attuazione}

\begin{frame}{Equazioni della Dinamica (Newton-Eulero)}
    Modello a 6 gradi di libertà per corpo rigido soggetto a forze e momenti esterni:
    
    \begin{block}{Equazioni del Moto nel Body Frame}
    \begin{equation}
        \begin{cases}
            m \bm{\dot{V}}_b + \bm{\Omega}_b \times (m \bm{V}_b) = \bm{F}_{\text{tot}} \\
            \mathbf{I}_b \bm{\dot{\Omega}}_b + \bm{\Omega}_b \times (\mathbf{I}_b \bm{\Omega}_b) = \bm{M}_{\text{tot}}
        \end{cases}
    \end{equation}
    \end{block}
    
    Scomposizione dei contributi:
    \begin{itemize}
        \item $\bm{F}_{\text{tot}} = \bm{F}_{\text{th}} + \bm{F}_g + \bm{F}_{\text{aero}}$
        \item $\bm{M}_{\text{tot}} = \bm{M}_{\text{th}} + \bm{M}_{\text{aero}} + \bm{M}_{\text{gyro}} + \bm{M}_{\text{pinna}}$
    \end{itemize}
\end{frame}

\begin{frame}{Scomposizione della Forza Totale ($F_{TOT}$)}
    La forza totale agente sul tricottero è la risultante di tre contributi principali:
    \begin{equation}
        F_{TOT} = F_{TH} + F_{GRAV} + F_{AERO}
    \end{equation}

    \begin{itemize}
        \item \textbf{Spinta ($F_{TH}$):} Generata dai tre rotori.
        \item \textbf{Gravità ($F_{GRAV}$):} Proiettata dal sistema inerziale al body frame tramite la matrice $R_{b \to g}$:
        \begin{equation}
            F_{GRAV} = \begin{bmatrix} -\sin(\theta) \\ \cos(\theta)\sin(\phi) \\ \cos(\theta)\cos(\phi) \end{bmatrix} mg
        \end{equation}
        \item \textbf{Aerodinamica ($F_{AERO}$):} Include Lift e Drag, dipendenti dal quadrato della velocità $v^2$ e dalla funzione segno $\text{sgn}(v)$ per la direzione.
    \end{itemize}
\end{frame}

\begin{frame}{Dinamica Traslazionale: Forze Aerodinamiche}
    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            \footnotesize
            Il Drag del corpo ($\mathbf{D}_{\text{Body}}$) agisce lungo gli assi principali:
            \begin{equation}
                \mathbf{D}_{\text{Body}} = -\frac{1}{2} \rho 
                \begin{bmatrix}
                    \text{sgn}(v_x) v_x^2 s_{b_x} C_{d_{b_x}} \\
                    \text{sgn}(v_y) v_y^2 s_{b_y} C_{d_{b_y}} \\
                    \text{sgn}(v_z) v_z^2 s_{b_z} C_{d_{b_z}}
                \end{bmatrix}
            \end{equation}
            
            Le ali generano Lift ($L$) e Drag ($D$):
            \begin{equation}
                \bm{F}_{aero,wing} \approx 
                \begin{bmatrix}
                    -\frac{1}{2}\rho s_x C_D(v_x) v_x |v_x| \\
                    0 \\
                    -\frac{1}{2}\rho s_z C_L(v_x) v_x^2
                \end{bmatrix}
            \end{equation}
        \end{column}

        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\linewidth]{forze_x.jpg}
            \captionof{figure}{\tiny Scomposizione: Lift e Drag.}
        \end{column}
    \end{columns}
\end{frame}

% --- SLIDE 1: COMPONENTI DELLE FORZE ---
\begin{frame}{3.3 Analisi delle Forze Esterne ($\bm{F}_{\text{tot}}$)}
    La dinamica traslazionale nel \textit{body frame} è determinata dalla risultante delle forze:
    \begin{equation*}
        m \bm{\dot{V}}_b = \bm{F}_{\text{th}} + \bm{F}_g + \bm{F}_{\text{aero}} - \bm{\Omega}_b \times (m \bm{V}_b)
    \end{equation*}

    \begin{itemize}
        \item \textbf{Forza di Gravità ($\bm{F}_g$):} Proiezione del peso tramite la matrice di rotazione trasposta.
        \begin{equation*}
            \bm{F}_g = \begin{bmatrix} -\sin\theta \\ \cos\theta \sin\phi \\ \cos\theta \cos\phi \end{bmatrix} mg
        \end{equation*}
        \item \textbf{Spinta Propulsiva ($\bm{F}_{\text{th}}$):} Risultante dei tre rotori, dove per i rotori anteriori si considera il tilt $\theta_1, \theta_2$:
        \begin{equation*}
            \bm{F}_{\text{th}} = \begin{bmatrix} 
                k\omega_1^2 \sin\theta_1 + k\omega_2^2 \sin\theta_2 \\ 
                k\omega_3^2 \cos\theta_3 \sin\theta_4 \\ 
                -(k\omega_1^2 \cos\theta_1 + k\omega_2^2 \cos\theta_2 + k\omega_3^2 \cos\theta_3 \cos\theta_4) 
            \end{bmatrix}
        \end{equation*}
    \end{itemize}
\end{frame}

% --- SLIDE 2: MATRICE DEI MOMENTI (PARTE 1) ---
\begin{frame}[allowframebreaks]{3.4.7 Il Momento Totale ($\bm{M}_{\text{tot}}$)}
    Il momento totale è la chiave del controllo d'assetto. Di seguito la scomposizione analitica dei contributi propulsivi e torcenti:

    \begin{block}{Momento dei Thrust ($M_{\text{th}}$) e Torcente ($M_{\text{tor}}$)}
        \scalebox{0.68}{$
        \begin{aligned}
        \bm{M}_{\text{th}} &= \begin{bmatrix}
            d_{m_y} k\omega_2^2 \sin\theta_2 - d_{m_y} k\omega_1^2 \sin\theta_1 - d_{t_y} k\omega_3^2 \sin\theta_3 - d_{t_z} k\omega_3^2 \cos\theta_3 \sin\theta_4 \\
            d_{m_z} k\omega_1^2 \cos\theta_1 + d_{m_z} k\omega_2^2 \cos\theta_2 + d_{m_x} k\omega_1^2 \sin\theta_1 + d_{m_x} k\omega_2^2 \sin\theta_2 + d_{t_x} k\omega_3^2 \sin\theta_3 + d_{t_z} k\omega_3^2 \cos\theta_3 \cos\theta_4 \\
            d_{m_y} k\omega_2^2 \cos\theta_2 - d_{m_y} k\omega_1^2 \cos\theta_1 - d_{t_y} k\omega_3^2 \cos\theta_3 \cos\theta_4 + d_{t_x} k\omega_3^2 \cos\theta_3 \sin\theta_4
        \end{bmatrix} \\
        \bm{M}_{\text{tor}} &= \begin{bmatrix}
            (b\omega_1^2 + I_{r12z} \dot{\omega}_1) \cos\theta_1 - (b\omega_2^2 + I_{r12z} \dot{\omega}_2) \cos\theta_2 + (b\omega_3^2 + I_{r3z} \dot{\omega}_3) \cos\theta_3 \cos\theta_4 \\
            (b\omega_3^2 + I_{r3z} \dot{\omega}_3) \cos\theta_3 \sin\theta_4 \\
            -(b\omega_1^2 + I_{r12z} \dot{\omega}_1) \sin\theta_1 + (b\omega_2^2 + I_{r12z} \dot{\omega}_2) \sin\theta_2 - (b\omega_3^2 + I_{r3z} \dot{\omega}_3) \sin\theta_3
        \end{bmatrix}
        \end{aligned}$}
    \end{block}

    \framebreak

    \begin{block}{Momento Aerodinamico ($\bm{M}_{\text{aero}}$) e Giroscopico ($\bm{M}_{\text{gyro}}$)}
        Il termine aerodinamico modella lo sfasamento tra Centro di Pressione e CoM, mentre quello giroscopico descrive le coppie indotte dalla variazione del tilt delle eliche.
    \end{block}

    \scalebox{0.7}{$
    \begin{aligned}
    \bm{M}_{\text{aero}} &\approx \begin{bmatrix}
        (l_{sx_y} + l_{dx_y}) \left( -\frac{1}{2} C_l \rho s v_x^2 - \frac{1}{2} C_{d_z} \rho s v_z^2 \text{sgn}(v_z) \right) \\
        -\left[ (l_{sx_x} + l_{dx_x}) \text{Lift} - (l_{sx_z} + l_{dx_z}) \text{Drag}_x \right] \\
        (l_{sx_y} + l_{dx_y}) \frac{1}{2} C_d \rho s v_x^2 \text{sgn}(v_x)
    \end{bmatrix} \\
    \bm{M}_{\text{gyro}} &= \begin{bmatrix}
        -I_{rz} \omega_{t1} \sin\theta_1 \omega_1 + I_{rz} \omega_{t2} \sin\theta_2 \omega_2 - I_{rz} \omega_{t3,xz} \sin\theta_3 \omega_3 - I_{ry} \omega_{t3,xy} \cos\theta_3 \sin\theta_4 \omega_3 \\
        I_{rx} \omega_{t3,xy} \cos\theta_3 \cos\theta_4 \omega_3 \\
        -I_{rz} \omega_{t1} \cos\theta_1 \omega_1 + I_{rz} \omega_{t2} \cos\theta_2 \omega_2 - I_{rz} \omega_{t3,xz} \cos\theta_3 \cos\theta_4 \omega_3
    \end{bmatrix}
    \end{aligned}$}
\end{frame}

% --- SLIDE 3: STABILIZZAZIONE PASSIVA ---
\begin{frame}{3.4.6 Stabilizzazione della Pinna e Accoppiamenti}
    Il modello si chiude con i termini di smorzamento passivo e i termini non lineari di Eulero:

    \begin{itemize}
        \item \textbf{Momento della Pinna ($\bm{M}_{\text{pinna}}$):} Modella la deriva di coda che stabilizza il drone in volo avanzato (funzione di $v_y^2$).
        \begin{equation*}
            \bm{M}_{\text{pinna}} = \begin{bmatrix} -p \alpha_{0x} - \alpha_{1x} v_y^2 \\ 0 \\ -r (\alpha_{0z} + \alpha_{1z} v_y^2) \end{bmatrix}
        \end{equation*}
        \item \textbf{Termini d'Inerzia (Coriolis):} Coppie dovute alla rotazione del corpo rigido.
        \begin{equation*}
            \bm{M}_{\text{inerz}} = \begin{bmatrix} (I_{zz} - I_{yy})qr \\ (I_{xx} - I_{zz})pr \\ (I_{yy} - I_{xx})pq \end{bmatrix}
        \end{equation*}
    \end{itemize}

    \begin{alertblock}{Conclusione sulla Dinamica}
        La presenza di termini proporzionali al quadrato della velocità ($v^2$) e funzioni discontinue ($\text{sgn}$) rende il sistema fortemente non lineare, richiedendo strategie di controllo robuste.
    \end{alertblock}
\end{frame}

\section{Architettura di Controllo}

\begin{frame}
    \vfill
    \centering
    {\Huge \textbf{Volo Verticale}}
    \vfill
\end{frame}

\begin{frame}{Configurazione fase di decollo}
    La prima fase di volo è quella di decollo verticale, durante la quale i motori sono posizionati verticalmente per generare una propulsione in grado di far sollevare il drone.
    
    \vspace{0.3cm}
    Si assume che in tale fase gli angoli di tilt siano i seguenti:
    \begin{itemize}
        \item $\theta_1 = \theta_2 = \pi/2$ (Rotori anteriori verticali).
        \item $\theta_4 = -\pi/2$ (Braccio di coda orientato lateralmente).
        \item $\theta_3 = \theta_{3,eq}$: tale angolo permette di contrastare il momento torcente attorno all'asse $z$ causato dalla coppia di reazione del motore posteriore, non essendo presente un secondo rotore controrotante per il bilanciamento.
    \end{itemize}
\end{frame}

\begin{frame}{Calcolo dell'Angolo di Equilibrio $\theta_{3_{eq}}$}
    \small
    In configurazione di \textbf{Hovering}, è necessario calcolare un angolo di tilt $\theta_3$ tale da annullare il momento di imbardata residuo generato dalla reazione del rotore di coda.

    \vspace{0.2cm}
    \textbf{1. Bilancio dei Momenti su Z ($M_{z}^{tot} = 0$):}
    \footnotesize
    \begin{align*}
        M_{th,z} &= \underbrace{d_{my} k (\omega_{2}^2 c_{\theta_2} - \omega_{1}^2 c_{\theta_1})}_{\approx 0 \text{ per } \theta_{1,2} \approx 90^\circ} 
                  - d_{ty} k \omega_{3}^2 c_{\theta_3} c_{\theta_4} 
                  + d_{tx} k \omega_{3}^2 c_{\theta_3} s_{\theta_4} \\
        M_{tor,z} &= \underbrace{-b \omega_{1}^2 s_{\theta_1} + b \omega_{2}^2 s_{\theta_2}}_{\approx 0 \text{ (Simmetria } \omega_1=\omega_2 \text{)}} 
                   - b \omega_{3}^2 s_{\theta_3}
    \end{align*}
    
    \vspace{0.1cm}
    \normalsize
    \textbf{2. Semplificazione e Soluzione:}
    Considerando $\theta_4$ fisso (orientamento laterale) e imponendo l'equilibrio:
    \begin{equation*}
        \omega_3^2 \left( d_{tx} k \cos\theta_3 \sin\theta_4 - b \sin\theta_3 \right) = 0
    \end{equation*}
    Dividendo per $\cos\theta_3$ e riarrangiando, si ottiene l'angolo di equilibrio statico:  
    \begin{equation}
        \boxed{\theta_{3_{eq}} = \arctan\left( -\frac{k d_{tx}}{b}\right)}
    \end{equation}
    
\end{frame}

% --- SLIDE 1: Panoramica Strategia ---
\begin{frame}{Architettura Gerarchica a Cascata}
    Sulla base dell'analisi dinamica, è stata implementata un'architettura di controllo a cascata che disaccoppia la dinamica lenta (posizione) da quella veloce (assetto).

    \begin{block}{Suddivisione Logica}
    \begin{enumerate}
        \item \textbf{Outer Loop (Posizione $x, y, z$):}
        Utilizza la tecnica \textbf{Sliding Mode Control (SMC)}.
        \begin{itemize}
            \item Calcola le forze richieste ($F_{req}$) per annullare l'errore di posizione.
            \item Effettua un \textit{mapping cinematico} traducendo le forze planari in angoli di assetto desiderati ($\phi_{des}, \theta_{des}$).
        \end{itemize}
        
        \item \textbf{Inner Loop (Assetto $\phi, \theta, \psi$):}
        Utilizza controllori \textbf{PID} per inseguire gli angoli desiderati calcolati dal loop esterno e stabilizzare le velocità angolari.
    \end{enumerate}
    \end{block}
    
    \vspace{0.2cm}
    \textbf{Peculiarità:} Il controllo di quota e quello planare sono \textbf{accoppiati}. La spinta totale $T$ (loop $z$) funge da termine di normalizzazione per determinare l'inclinazione del vettore spinta.
\end{frame}

\begin{frame}{Analisi di Stabilità (Metodo di Lyapunov)}
    
    Per garantire la stabilità asintotica globale della \textit{sliding surface} ($\sigma \to 0$), si utilizza il criterio di Lyapunov.

    \vspace{0.3cm}
    
    \begin{columns}[T]
        % --- COLONNA SINISTRA: I Passaggi Logici ---
        \begin{column}{0.6\textwidth}
            \textbf{1. Funzione Candidata:}
            \begin{equation*}
                V(\sigma) = \frac{1}{2}\sigma^2 > 0
            \end{equation*}
            
            \textbf{2. Derivata Temporale:}
            Sostituendo la dinamica a ciclo chiuso con disturbo $\Delta$:
            \begin{align*}
                \dot{V} &= \sigma \dot{\sigma} \\
                        &= -k \sigma \text{sgn}(\sigma) + \sigma \Delta \\
                        &\le -k |\sigma| + |\sigma| \delta_{\max}
            \end{align*}
        \end{column}
        
        % --- COLONNA DESTRA: La Condizione ---
        \begin{column}{0.4\textwidth}
            \centering
            \vspace{1cm}
            \begin{alertblock}{Condizione di Stabilità}
                Affinché la funzione decresca ($\dot{V} < 0$), il guadagno deve "battere" il disturbo:
                \vspace{0.2cm}
                \centering
                \large
                \boldmath
                $k > \delta_{\max}$
                \unboldmath
            \end{alertblock}
            
            \vspace{0.3cm}
            \small \itshape
            Il sistema è robusto se il guadagno di switching supera il limite superiore dell'incertezza.
        \end{column}
    \end{columns}

    % Grafico concettuale opzionale in basso
    \vspace{0.5cm}
    \centering
    \begin{tikzpicture}[scale=0.6]
        \draw[->] (0,0) -- (6,0) node[right] {$t$};
        \draw[->] (0,0) -- (0,2.5) node[above] {$V(\sigma)$};
        \draw[thick, blue] plot [domain=0:5.5, samples=100] (\x, {2*exp(-\x)});
        \node[blue] at (2, 1.5) {$\dot{V} < 0$};
        \draw[dashed] (0,0) node[left] {0} -- (6,0);
    \end{tikzpicture}

\end{frame}

\begin{frame}{Gestione dei Disturbi e Tuning del Guadagno}

    \begin{columns}[c]

            \textbf{Condizione di Robustezza:}
            $$ k > |\Delta(t)|_{\max} $$

\begin{frame}{Controllo Planare e Inversione Dinamica}
    Essendo il sistema \textbf{sotto-attuato} sugli assi $x, y$, si utilizza la spinta totale $T$ (calcolata dal loop $z$) per generare le componenti orizzontali tramite il rollio ($\phi$) e il beccheggio ($\theta$).
    
    \vspace{0.2cm}
    Gli SMC planari calcolano le "forze virtuali" ($F_{y,req}, F_{x,req}$), che vengono convertite in angoli di riferimento invertendo la dinamica:
    
    \begin{columns}[t]
        \column{0.48\textwidth}
        \centering
        \textbf{Generazione Rollio ($\phi_{des}$)}
        \footnotesize
        \begin{equation}
            \phi_{des} = \arcsin\left( \frac{F_{y,req}}{T} \right)
        \end{equation}
        \raggedright
        Compensa l'errore laterale $e_y$.

        \column{0.48\textwidth}
        \centering
        \textbf{Generazione Beccheggio ($\theta_{des}$)}
        \footnotesize
        \begin{equation}
            \theta_{des} = \arcsin\left( - \frac{F_{x,req}}{T} \right)
        \end{equation}
        \raggedright
        Compensa l'errore longitudinale $e_x$.
    \end{columns}
    
    \vspace{0.3cm}
    \begin{alertblock}{Ipotesi di Validità}
    L'inversione assume \textbf{piccoli angoli} ($\sin\theta \approx \theta$) e una \textbf{separazione delle scale temporali}: la dinamica di assetto (inner loop) deve essere sufficientemente più veloce di quella di posizione (outer loop).
    \end{alertblock}
\end{frame}

\begin{frame}{Controllo di Assetto (Inner Loop)}
Per inseguire gli angoli di riferimento ($\phi_{des}, \theta_{des}, \psi_{des}$) è necessario calcolare dei momenti torcenti rispettivamente lungo gli assi $x, y, z$.
Tali quantità sono calcolate come uscite di tre PID.
\begin{itemize}
    \item \textbf{Roll:} è stato applicato un controllo Proporzionale-Derivativo
    \begin{equation}
        M_{x,req} = k_{p} (\phi_{des} - \phi) + k_{d} (\dot{\phi}_{des} - p)
    \end{equation}
    \item \textbf{Pitch:} è stato applicato un controllo Proporzionale-Integrale-Derivativo. Il termine integrale reagisce alla condizione in cui il centro di pressione non coincida perfettamente con il centro di massa.
    \begin{equation}
        M_{y,req} = k_{p} (\theta_{des} - \theta) + k_{i} \int (\theta_{des} - \theta) \,dt\ + k_{d} (\dot{\theta}_{des} - q)
    \end{equation}
    \item \textbf{Yaw:} è stato applicato un controllo Proporzionale-Derivativo
    \begin{equation}
        M_{z,req} = k_{p} (\psi_{des} - \psi) + k_{d} (\dot{\psi}_{des} - r)
    \end{equation}
\end{itemize}
\end{frame}

\begin{frame}{Mixer: Quota e Pitch}
    Il controllo dell'equilibrio longitudinale richiede la risoluzione di un sistema accoppiato: ogni variazione di spinta per la quota ($F_{z,req}$) induce un momento di beccheggio, e viceversa.
    
    \vspace{0.2cm}
    Tramite il mixer si determina la ripartizione ottima della spinta tra i rotori anteriori e il rotore posteriore:
    \begin{equation}
        \begin{cases}
        T_{front} + k \omega_{3}^2 \sin(\theta_{3}) = F_{z,req} \\
        T_{front} d_{mx} + k \omega_{3}^2 \sin(\theta_{3}) d_{tx} + b \omega_{3}^2 \cos(\theta_{3}) \sin(\theta_{4}) = M_y
        \end{cases}
    \end{equation}
    
    \vspace{0.2cm}
    Esprimendo $T_{front}$ come $2 k \omega_{i}^2$ (con $i=1,2$):
    \begin{equation}
        \begin{cases}
        2 k \omega_{i}^2 + k \omega_{3}^2 \sin(\theta_3) = F_{z,req} \\
        2 k \omega_{i}^2 d_{mx} + k \omega_{3}^2 \sin(\theta_3) d_{tx} + b \omega_{3}^2 \cos(\theta_{3}) \sin(\theta_{4}) = M_{y}
        \end{cases}
    \end{equation}
\end{frame}

\begin{frame}{Mixer: Quota e Pitch}
    Risolvendo il sistema per l'incognita $\omega_{3}^2$, si ricava la velocità angolare del rotore di coda:
    \begin{equation}
        \omega_3^2 = \frac{M_y - F_{z,req} d_{mx}}{k \sin(\theta_3) (d_{tx} - d_{mx}) + b \cos(\theta_3) \sin(\theta_4)}
    \end{equation}
    
    \vspace{0.3cm}
    Una volta determinato $T_3 = k \omega_3^2$, si calcola la spinta necessaria per il gruppo propulsivo anteriore ($T_{front}$) per garantire il sostentamento e l'assetto richiesto:
    \begin{equation}
        T_{front} = F_{z,req} - T_3 \sin(\theta_3)
    \end{equation}
    
    \vspace{0.3cm}
\end{frame}

\begin{frame}{Mixer: Roll}
    \textbf{2. Equilibrio Laterale (Roll):}
    Il rollio viene raggiunto attraverso la spinta differenziale dei rotori anteriori.\\
    La spinta totale anteriore $T_{front}$ viene ripartita tra il motore sinistro (2) e destro (1) per generare il momento di rollio richiesto ($M_x$).
    Il sistema da risolvere è:
    \begin{equation}
        \begin{cases}
        T_1 + T_2 = T_{front} \\
        (T_2 - T_1) d_{my} = M_x
        \end{cases}
    \end{equation}
    Definendo la spinta media anteriore come $T_i = T_{front} / 2$, le spinte dei singoli motori si ottengono per somma e differenza rispetto al termine differenziale legato al momento:

    \begin{equation}
        T_1 = T_i - \frac{M_x}{2 d_{my}}, \quad T_2 = T_i + \frac{M_x}{2 d_{my}}
    \end{equation}
    Le rispettive velocità angolari si ricavano infine come $\omega_{1,2} = \sqrt{T_{1,2}/k}$.
\end{frame}

\begin{frame}{Mixer: Yaw}
    \textbf{3. Controllo Imbardata (Yaw):}
    L'imbardata è controllata attraverso il tilt differenziale dei rotori anteriori ($\delta$).
    La relazione che lega il momento torcente $M_z$ all'angolo di inclinazione è approssimabile come:
    \begin{equation}
        M_z = T_{front} d_{my} \sin(\delta)
    \end{equation}
    
    \vspace{0.2cm}
    Invertendo la relazione, si calcola l'angolo di tilt desiderato con saturazione di sicurezza ($\delta_{max}$):
    \begin{equation}
        \delta_{des} = \arcsin\left( \frac{M_{z,req}}{T_{front} d_{my}} \right)
    \end{equation}
    
    \vspace{0.3cm}
    \footnotesize
    \begin{itemize}
        \item \textit{Compensazione:} Il controllo bilancia la coppia residua del rotore 3 ($\theta_{3,eq}$).
        \item \textbf{Gestione Saturazione:} Se il controllo richiede $\delta > \delta_{max}$, l'imbardata perde priorità per non compromettere la stabilità di rollio e beccheggio (sicurezza del volo).
    \end{itemize}
\end{frame}

\begin{frame}{Strategia di Controllo Orizzontale}
    Nella fase di volo orizzontale (Fase 3), il rotore di coda viene disattivato ($T_{rear} = 0$). 
    Il sostentamento è affidato alla portanza alare e il controllo avviene tramite \textbf{Thrust Vectoring} dei rotori anteriori.
    
    \vspace{0.3cm}
    \begin{block}{Obiettivi del Controllore (Case 12)}
        \begin{itemize}
            \item \textbf{Longitudinale:} Inseguimento velocità $v_{x,des}$ e quota $z_{des}$.
            \item \textbf{Laterale:} Mantenimento rotta ($y=0$) tramite correzione dell'imbardata (Yaw-steering).
            \item \textbf{Assetto:} Stabilizzazione attiva di Roll e Pitch a $0^\circ$.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Architettura a Cascata e Feedforward}
    Il controllo utilizza un approccio vettoriale: i loop esterni calcolano le forze richieste ($F_x, F_z$), che vengono convertite in intensità di spinta e angolo di tilt.

    \vspace{0.2cm}
    \textbf{Feedforward Aerodinamico:}
    Per migliorare la risposta, il controllore compensa le forze aerodinamiche note:
    \begin{itemize}
        \item \textbf{Lift (Ala):} $L_{ff} = \frac{1}{2} \rho S C_L v^2$. Riduce lo sforzo verticale dei motori all'aumentare della velocità.
        \item \textbf{Drag:} $D_{ff} = \frac{1}{2} \rho (S C_D + S_{body} C_{D_{body}}) v^2$. Anticipa la spinta necessaria per mantenere la velocità.
    \end{itemize}
    
    \begin{equation}
        \begin{cases}
            F_{z,req} = m(g - a_{z,PID}) - L_{ff} \\
            F_{x,req} = D_{ff} + F_{x,PID}
        \end{cases}
    \end{equation}
\end{frame}

\begin{frame}{Schema a Blocchi: Volo Orizzontale}
    \begin{figure} % AGGIUNTO FIGURE
        \centering

            \tikzstyle{input} = [coordinate] % AGGIUNTO STILE INPUT MANCANTE

            \node [sum, right of=guidance, node distance=2.5cm] (sum_ff) {$\Sigma$};

\begin{frame}{Strategia di Thrust Vectoring}
    La peculiarità del case 12 è il disaccoppiamento tra l'orientamento del vettore spinta e l'assetto del drone.
    
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{1. Calcolo Vettore Ideale:}
        L'angolo di spinta richiesto nel sistema globale è:
        \begin{equation}
            \alpha_{ideal} = \arctan\left(\frac{F_{z,req}}{F_{x,req}}\right)
        \end{equation}
        
        \textbf{2. Compensazione Pitch:}
        L'angolo effettivo dei servomotori deve tener conto dell'inclinazione corrente del corpo ($\theta$):
        \begin{equation}
            \alpha_{servo} = \text{sat}(\alpha_{ideal}) - \theta
        \end{equation}
        
        \column{0.45\textwidth}
        \centering
        % Qui potresti mettere un'immagine vettoriale se ce l'hai, altrimenti descrittivo
        \begin{alertblock}{Nota Bene}
            Se il drone alza il muso ($\theta > 0$), i servi ruotano verso il basso per mantenere il vettore spinta orizzontale e garantire la propulsione.
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}{Mixer e Allocazione Attuatori}
    I comandi di assetto (Inner Loop) vengono miscelati con il vettore di spinta principale:
    
    \begin{itemize}
        \item \textbf{Motori (Spinta Differenziale):} Gestiscono l'imbardata ($M_z$).
        \begin{equation*}
            T_{1,2} = \frac{T_{tot}}{2} \mp u_{yaw}
        \end{equation*}
        \item \textbf{Servi (Tilt Differenziale):} Gestiscono il Rollio ($M_x$) e correggono il Pitch ($M_y$).
        \begin{equation*}
            \begin{cases}
                \delta_{servo,1} = \alpha_{servo} + u_{pitch} - u_{roll} \\
                \delta_{servo,2} = \alpha_{servo} + u_{pitch} + u_{roll}
            \end{cases}
        \end{equation*}
    \end{itemize}
    
    \footnotesize
    \textit{*Il rotore di coda è spento ($u_3 = 0$) per massimizzare l'efficienza.}
\end{frame}
