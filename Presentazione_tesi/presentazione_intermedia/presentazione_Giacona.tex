\begin{frame}{VTOL tilting-rotor}
    Il tricottero \textit{tilting-rotor} combina i benefici delle due architetture:
    
    \begin{columns}
        \column{0.6\textwidth}

        \begin{bmatrix} \bm{V}_b \\ \bm{\Omega}_b \end{bmatrix}
    \end{equation}
\end{frame}

% --- SLIDE 3: NEWTON-EULERO ---
\begin{frame}{Equazioni della Dinamica: Newton-Eulero}
    Il modello del corpo rigido è descritto dalle equazioni cardinali nel \textit{Body Frame}:
    
    \vspace{0.5cm}
    \begin{block}{Dinamica Traslazionale e Rotazionale}
    \begin{equation}
        \begin{cases}
            m \bm{\dot{V}}_b = \bm{F}_{\text{TOT}} - \underbrace{\bm{\Omega}_b \times (m \bm{V}_b)}_{\text{Coriolis}} \\
            \mathbf{I}_b \bm{\dot{\Omega}}_b = \bm{M}_{\text{TOT}} - \underbrace{\bm{\Omega}_b \times (\mathbf{I}_b \bm{\Omega}_b)}_{\text{Giroscopico Body}}
        \end{cases}
    \end{equation}
    \end{block}
    
    \vspace{0.5cm}
    \textbf{Scomposizione delle Forze e Momenti:}
    \begin{equation}
        \bm{F}_{\text{TOT}} = \bm{F}_{\text{TH}} + \bm{F}_{\text{GRAV}} + \bm{F}_{\text{AERO}}
    \end{equation}
    \begin{equation}
        \bm{M}_{\text{TOT}} = \bm{M}_{\text{TH}} + \bm{M}_{\text{TOR}} + \bm{M}_{\text{AERO}} + \bm{M}_{\text{GIRO}} + \bm{M}_{\text{PINNA}}
    \end{equation}
    
\end{frame}

% --- SLIDE 4: FORZE ESTERNE ---
\begin{frame}{Analisi delle Forze}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{1. Spinta Motori} \\
            Dipende dall'orientamento ($\theta_i$) e velocità ($\omega_i$) dei rotori:
            \begin{equation}
                \bm{F}_{\text{TH}} = \sum_{i=1}^3 \mathbf{R}_{rot_i \to b}(\theta_i) \cdot [0, 0, k\omega_i^2]^T
            \end{equation}
            
            \vspace{0.2cm}
            \textbf{2. Gravità} \\
            Proiezione del vettore peso nel body frame:
            \begin{equation}
                \bm{F}_{\text{GRAV}} = \mathbf{R}_{g \to b} \cdot [0, 0, mg]^T
            \end{equation}
        \end{column}
        
        \begin{column}{0.5\textwidth}
            \textbf{3. Aerodinamica} \\
            Modello non-lineare per fusoliera e ali:
            \begin{equation}
                \bm{F}_{\text{AERO}} = \bm{F}_{wing} + \bm{D}_{body}
            \end{equation}
            \vspace{0.1cm}
            \centering
            \includegraphics[width=0.68\linewidth]{forze_x.jpg}
            \captionof{figure}{Drag e Lift.}
        \end{column}
    \end{columns}
\end{frame}


\begin{frame}{Modellistica dei Momenti: Attivi e Reattivi}
    \small
    \vspace{-0.2cm} % Recupera spazio in alto
    
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Momenti di Controllo (Attivi)}

                \begin{itemize}
                    \item \textbf{Spinta:} 
                    \begin{equation}
                        \bm{M}_{\text{TH}} = \textstyle \sum (\bm{r}_i \times \bm{F}_{th,i})
                    \end{equation}
                    \item \textbf{Coppia:} 
                    \begin{equation}
                        \bm{M}_{\text{TOR}} = \textstyle \sum (b \omega_i^2 + I_{r} \dot{\omega}_i) \hat{e}_{i}
                    \end{equation}
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Dinamica Veloce (Reattivi)}

                \begin{itemize}
                    \item \textbf{Giroscopico Rotori:}
                    \begin{equation}
                        \bm{M}_{\text{GIRO}} = \textstyle \sum \bm{\omega}_{t,i} \times (J_r \bm{\omega}_{r,i})
                    \end{equation}
                    \item \textbf{Giroscopico Body:}
                    \begin{equation}
                        \bm{M}_{body} = \bm{\Omega}_b \times (\mathbf{I}_b \bm{\Omega}_b)
                    \end{equation}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

                \end{equation}
            \end{exampleblock}

% --- SLIDE 1: FORMULAZIONE DELLO STATO ---

\begin{frame}{Formulazione nello Spazio di Stato}
    Il modello è tradotto in un sistema di ODE del tipo $\dot{\mathbf{x}} = f(\mathbf{x}, \mathbf{u}, t)$.
    
    \vspace{0.5cm}
    \textbf{Vettore di Stato ($\mathbf{x} \in \mathbb{R}^{26}$):}
    \begin{equation}
        \mathbf{x} = 
        {
            \renewcommand{\arraystretch}{1.2}
        \begin{bmatrix}
            \bm{P}_g \\ \bm{V}_b \\ \hline
            \bm{\Theta}_g \\ \bm{\Omega}_b \\ \hline
            \bm{\theta}_{rot} \\ \bm{\dot{\theta}}_{rot} \\ \hline
            \bm{\omega}_{rot} \\ \bm{\dot{\omega}}_{rot}
        \end{bmatrix}
        }
        \begin{array}{l}
            \left. \begin{matrix} \text{Posizione Inerziale } [x,y,z] \\ \text{Velocità Body } [u,v,w] \end{matrix} \right\} \text{\textbf{Traslazione}} \\[1em]
            \left. \begin{matrix} \text{Assetto Eulero } [\phi,\theta,\psi] \\ \text{Velocità Angolari } [p,q,r] \end{matrix} \right\} \text{\textbf{Rotazione}} \\[1em]
            \left. \begin{matrix} \text{Posizione Servi} \\ \text{Velocità Servi} \end{matrix} \right\} \text{\textbf{Dinamica Servi}} \\[1em]
            \left. \begin{matrix} \text{Velocità Rotori} \\ \text{Accel. Rotori} \end{matrix} \right\} \text{\textbf{Propulsione}}
        \end{array}
    \end{equation}
\end{frame}

% --- SLIDE 2: INGRESSI DI CONTROLLO ---
% Più pulita, usa colonne per separare logica e matematica
\begin{frame}{Ingressi di Controllo ($\mathbf{u} \in \mathbb{R}^7$)}
    Il vettore degli ingressi $\mathbf{u}$ gestisce sia la potenza che la geometria del velivolo.

    \vspace{0.5cm}
    \begin{columns}[c]
        \begin{column}{0.4\textwidth}
            \centering
            \begin{equation}
                \mathbf{u} = \begin{bmatrix} 
                u_1 \\ u_2 \\ u_3 \\ \hline
                u_4 \\ u_5 \\ \hline
                u_6 \\ u_7 
                \end{bmatrix}
            \end{equation}
        \end{column}

        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item[] \textbf{Propulsione}
                \begin{itemize}
                    \item[$u_{1-3}$:] Velocità angolari per i tre motori.
                \end{itemize}
                \vspace{0.3cm}
                \item[] \textbf{Tilt Anteriore}
                \begin{itemize}
                    \item[$u_{4,5}$:] Angoli di tilt per rotori anteriori.
                \end{itemize}
                \vspace{0.3cm}
                \item[] \textbf{Tilt Coda}
                \begin{itemize}
                    \item[$u_{6,7}$:] Angoli di tilt rotore di coda.
                \end{itemize}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Sintesi della Dinamica del Sistema}
    \vspace{-0.2cm}
    Il modello dello stato $\mathbf{x} \in \mathbb{R}^{26}$ è governato dalle equazioni di Newton-Euler espresse nel \textit{body frame} ($B$) e proiettate nel sistema inerziale ($G$).
    
    \begin{block}{Cinematica: Relazione $B \to G$}
        \centering
        $\begin{bmatrix} \dot{\mathbf{p}}_g \\ \dot{\boldsymbol{\eta}}_g \end{bmatrix} = 
        \begin{bmatrix} R_{b \to g}(\boldsymbol{\eta}) & \mathbb{O}_{3\times3} \\ \mathbb{O}_{3\times3} & J_{b \to g}(\boldsymbol{\eta}) \end{bmatrix} 
        \begin{bmatrix} \mathbf{v}_b \\ \boldsymbol{\omega}_b \end{bmatrix}$
    \end{block}

    \begin{block}{Dinamica nel Body Frame}
        \centering
        $ \dot{\mathbf{v}}_b = \frac{1}{m} \mathbf{F}_{TOT} - \boldsymbol{\omega}_b \times \mathbf{v}_b $ \\
        $ \dot{\boldsymbol{\omega}}_b = I_B^{-1} \left( \mathbf{M}_{TOT} - \boldsymbol{\omega}_b \times (I_B \boldsymbol{\omega}_b) \right) $
    \end{block}

    \begin{exampleblock}{Contributi alle Azioni Generalizzate}
        \small
        \begin{itemize}
            \item \textbf{Forze:} $\mathbf{F}_{TH}$ (Thrust motori $x_{21,23,25}$), $\mathbf{F}_{GRAV}(\theta, \phi)$, $\mathbf{F}_{AERO}$ (Drag/Lift).
            \item \textbf{Momenti:} Bracci motori ($d$), Effetti Giroscopici ($I_{rot} \omega \Omega$), Smorzamenti aerodinamici ($\alpha$) e Coppie di trascinamento ($b \omega^2$).
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{Configurazione di Hovering}
    In fase di decollo, i rotori anteriori sono verticali ($\theta_{1,2} = \pi/2$).
    
    \vspace{0.3cm}
    \begin{columns}[c]
        \begin{column}{0.56\textwidth}
            \textbf{Problema:} Il rotore di coda genera una coppia di reazione non bilanciata da un controrotore.
            
            \vspace{0.2cm}
            \textbf{Soluzione ($\theta_{3,eq}$):}
            Si inclina il rotore di coda per generare una componente di spinta laterale che annulli il momento torcente su $Z$.
            
            \vspace{0.2cm}
            \begin{alertblock}{Angolo di Equilibrio}
            Imponendo $M_{TH,z} + M_{TOR,z} = 0$:
            \begin{equation}
                \theta_{3_{eq}} = \arctan\left( -\frac{k \, d_{tx}}{b}\right)
            \end{equation}
            \end{alertblock}
        \end{column}
        
        \begin{column}{0.4\textwidth}

    \vspace{0.5cm}

% --- SLIDE 4: SMC QUOTA (INVARIATA) ---
\begin{frame}{Outer Loop: Controllo di Quota}

    \vspace{0.2cm}
    \textbf{1. Definizioni:}
    \begin{itemize}
        \item Errore: $e_z = z_{des} - z$
        \item Sliding surface: $\sigma_z = \dot{e}_z + \lambda_z e_z$, con $\lambda > 0$
    \end{itemize}
    \vspace{0.3cm}
    Si considera l'equazione della dinamica lungo l'asse $z$ inerziale:
    \begin{equation}
        m\ddot{z} = mg - \cos(\phi)\cos(\theta) F_{z,req} + \cos(\phi)\cos(\theta) F_{aero,z}
    \end{equation}
    Imponendo la \textit{reaching law}: $\dot{\sigma} = -k \tanh\left( \frac{\sigma_z}{\Phi} \right)$.
    \vspace{0.1cm}
    \\Genera la richiesta di Thrust ($F_{z,req}$) per portare il drone in quota e mantenerlo in hovering:
    \begin{equation}
        F_{z,req} = \frac{m}{\cos(\phi) \cos(\theta)} \left(g - \ddot{z}_{des} - \lambda_z \dot{e}_z - K_z \tanh\left(\frac{\sigma_z}{\Phi}\right) \right) + F_{aero,z}
    \end{equation}
    

\end{frame}


% --- SLIDE 1: Fondamenti e Stabilità ---
\begin{frame}{Analisi di Stabilità per il Controllo di Quota}
    \begin{block}{Obiettivo del Controllo}
        Forzare lo stato sulla superficie $\sigma = 0$ in \textbf{tempo finito} e garantire la convergenza dell'errore.
    \end{block}

    \textbf{Analisi di Lyapunov:}
    Sia $V = \frac{1}{2}\sigma^2$ e il controllo $u = -k \operatorname{sgn}(\sigma)$.\\
    \vspace{0.05cm}
    La dinamica perturbata è $\dot{\sigma} = -k \operatorname{sgn}(\sigma) - \Delta(t)$, con $|\Delta(t)| \le \Delta_{max}$.
    
    \begin{equation}
        \dot{V} = \sigma\dot{\sigma} \le -(k - \Delta_{max})|\sigma| = -\delta \sqrt{2V}
    \end{equation}

    \vspace{0.2cm}
    \begin{columns}[t]
        \begin{column}{0.45\textwidth}
            \textbf{1. Robustezza} \\
            Garantita se il guadagno soddisfa:
            \begin{equation}
                k > \Delta_{max}
            \end{equation}
        \end{column}
        \begin{column}{0.45\textwidth}
            \textbf{2. Tempo di Raggiungimento} \\
            Integrando la disuguaglianza:
            \begin{equation}
                t_r \le \frac{|\sigma(0)|}{k - \Delta_{max}}
            \end{equation}
        \end{column}
    \end{columns}
\end{frame}

% --- SLIDE 2: Regolarizzazione (Chattering) ---
\begin{frame}{Regolarizzazione e Boundary Layer}
    \begin{alertblock}{Il Problema del Chattering}
        La discontinuità di $\operatorname{sgn}(\sigma)$ causa oscillazioni ad alta frequenza dannose per gli attuatori.
    \end{alertblock}

    Si introduce un \textbf{boundary layer} $B = \{x \in \mathbb{R}^n: |\sigma(x)| \leq \Phi\}$, definendo il controllo:
    \begin{equation}
        u = -k \tanh\left( \frac{\sigma}{\Phi} \right)
    \end{equation}

    Affinché la regione sia attrattiva, deve valere $\dot{V} < 0$ sulla frontiera $|\sigma| = \Phi$:
    \begin{equation}
        \dot{V} \le -\Phi \left( k \tanh(1) - \Delta_{max} \right) < 0
        \implies k > \frac{\Delta_{max}}{\tanh(1)}
    \end{equation}

    \begin{block}{Finite Time Practical Stability}
        Il sistema non converge più asintoticamente a zero, ma entra in tempo finito nel \textit{boundary layer}. Si parla di stabilità pratica, perché dipende dal tuning dei parametri $k$ e $\Phi$.
    \end{block}
\end{frame}


% --- SLIDE 7: INVERSIONE CINEMATICA (IL PONTE) ---
\begin{frame}{Inversione Cinematica: Dal Loop Esterno all'Interno}
    Le forze virtuali generate dagli SMC di posizione devono essere tradotte in angoli di assetto.
    
    \vspace{0.4cm}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{block}{Generazione Rollio ($\phi_{des}$)}
                \begin{equation}
                    F_{y,req} = m \left( K_y \tanh\left(\frac{\sigma_y}{\Phi}\right) + \lambda_y \dot{e}_y \right)
                \end{equation}
                \begin{equation}
                    \phi_{des} = \arcsin\left( \frac{F_{y,req}}{T} \right)
                \end{equation}
            \end{block}
        \end{column}
        
        \hfill 
        
        \begin{column}{0.48\textwidth}
            \begin{block}{Generazione Beccheggio ($\theta_{des}$)}
                \begin{equation}
                    F_{x,req} = -m \left(K_x \tanh\left(\frac{\sigma_x}{\Phi}\right) + \lambda_x \dot{e}_x \right)
                \end{equation}
                \begin{equation}
                    \theta_{des} = \arcsin\left( - \frac{F_{x,req}}{T} \right)
                \end{equation}
            \end{block}
        \end{column}
    \end{columns}
    
    \vspace{0.4cm}
    \begin{itemize}
        \item \textbf{Input:} Forze $F_{req}$ (SMC esterno).
        \item \textbf{Output:} Riferimenti angolari $\mathbf{A}_{des}$ per l'SMC interno.
    \end{itemize}
\end{frame}

% --- SLIDE 8: INNER LOOP SMC (IL MOTORE DELLA STABILITÀ) ---

\begin{frame}{Inner Loop: Controllo d'Assetto}
    L'obiettivo è forzare l'errore d'assetto sulla \textbf{sliding surface}, dove il disturbo viene annullato dalla commutazione del controllo.
    \begin{itemize}
    \item Definiamo la \textit{sliding surface}:
    \begin{equation*}
        \sigma = \dot{e} + \lambda e \quad \longrightarrow \quad \text{se } \sigma=0, \text{ l'errore decade esponenzialmente.}
    \end{equation*}

    \item Attraverso lo SMC vengono generati dei momenti lungo gli assi $x, y, z$ 
    con i quali si raggiungono gli angoli desiderati.

    \end{itemize}
    \begin{align}
        M_{x,req} = I_{xx} \left[ \ddot{\phi}_{des} + \lambda_{\phi} \dot{e}_\phi + K_{\phi} \tanh\left( \frac{\sigma_\phi}{\Phi_{\phi}} \right) \right] - (I_{yy} - I_{zz})qr \\
        M_{y,req} = I_{yy} \left[ \ddot{\theta}_{des} + \lambda_{\theta} \dot{e}_\theta + K_{\theta} \tanh\left( \frac{\sigma_\theta}{\Phi_{\theta}} \right) \right] - (I_{zz} - I_{xx})pr\\
        M_{z,req} = I_{zz} \left[ \ddot{\psi}_{des} + \lambda_\psi \dot{e}_\psi + K_\psi \tanh\left( \frac{\sigma_\psi}{\Phi_{\psi}} \right) \right] - (I_{xx} - I_{yy})pq
    \end{align}

\end{frame}


\begin{frame}{Motor Mixing Algorithm: Thrust e Imbardata}
    Inizialmente, si ipotizza di ripartire equamente la spinta $F_{z,req}$ fra i tre motori.
    \begin{equation}
        \begin{cases}
            T_{front} = \frac{2}{3}F_{z,req}\\
            T_{3} = \frac{F_{z,req}}{3}
        \end{cases}          
    \end{equation}
    con $T_{front} = T_1 + T_2$.\\
    Successivamente il thrust viene distribuito tra i motori anteriori e posteriore in modo da inseguire l'angolo di pitch.

    \vspace{0.2cm}

    Il controllo dell'imbardata è gestito tramite il tilting differenziale dei rotori anteriori. 
    \\Si usa la relazione tra momento richiesto e angolo di inclinazione:
    \begin{equation}
        M_z = T_{front} d_{my} \sin(\delta)
    \end{equation}
    \begin{equation}
        \delta = \arcsin\left( \frac{M_z}{T_{front} d_{my}} \right)
        \implies 
        \begin{cases}
            \theta_1 = \pi/2 - \delta \\
            \theta_2 = \pi/2 + \delta
        \end{cases}
    \end{equation}
\end{frame}

% --- SLIDE 8: MMA Longitudinale (Condensata) ---
\begin{frame}{Motor Mixing Algorithm: Quota e Beccheggio}
    La dinamica longitudinale richiede la risoluzione di un sistema con Spinta ($F_{z,req}$) e Beccheggio ($M_y$).
    
    \begin{equation}
        \begin{cases}
            (T_1 + T_2) \cos(\delta) + T_3 \sin(\theta_{3}) = F_{z} \\
            (T_1 + T_2) \cos(\delta) d_{mx} + T_3 \sin(\theta_{3}) d_{tx} + M_{TOR,3} = M_y
        \end{cases}
    \end{equation}

    \vspace{0.3cm}
    \textbf{1. Calcolo Rotore di Coda ($T_3$):}
    Dall'equilibrio dei momenti attorno a Y, si isola il contributo della coda:
    \begin{equation}
        T_3 = k \left( \frac{M_y - F_{z} d_{mx}}{k \sin(\theta_3) (d_{mx} - d_{tx}) - b \cos(\theta_3)\sin(\theta_4)} \right)
    \end{equation}
    
    \vspace{0.3cm}
    \textbf{2. Calcolo Rotori Anteriori ($T_{front}$):}
    Nota la spinta della coda, il resto del sostentamento è affidato ai rotori anteriori:
    \begin{equation}
        T_{front} = \frac{F_{z} - T_3 \sin(\theta_3)}{\cos(\delta)}
    \end{equation}
\end{frame}

% --- SLIDE 9: MMA Laterale (Condensata) ---
\begin{frame}{Motor Mixing Algorithm: Rollio}
    Una volta determinato $T_{front}$, questo viene modulato per controllare il Rollio, che si ottiene tramite spinta differenziale dei rotori anteriori.
    \\Il sistema da risolvere è:
    \begin{equation}
        \begin{cases}
        T_1 + T_2 = T_{front} \\
        (T_2 - T_1) \cos(\delta) d_{my} = M_x
        \end{cases}
        \implies
        \begin{cases}
        T_1 + T_2 = T_{front} \\
        \Delta T \cos(\delta) d_{my} = M_x
        \end{cases}
    \end{equation}
    Da cui si ottiene il valore differenziale della spinta:
    \begin{equation}
        \Delta T = \frac{M_{x,req}}{2 d_{my} \cos(\delta)}
    \end{equation}
    I valori di Thrust per i motori anteriori diventano:
    \begin{equation}
        \begin{cases}
            T_{1} = \frac{T_{front}}{2} - \Delta T\\
            T_{2} = \frac{T_{front}}{2} + \Delta T
        \end{cases}
    \end{equation}           
\end{frame}

\begin{frame}{Ingressi di controllo}
    I comandi dati agli attuatori sono quindi i seguenti:
    \begin{gather}
        u_1 = \omega_1 = \sqrt{\frac{1}{k} \left(\frac{T_{front}}{2} - \Delta T \right)}\\
        u_2 = \omega_2 = \sqrt{\frac{1}{k} \left(\frac{T_{front}}{2} + \Delta T \right)}\\
        u_3 = \omega_3 = \sqrt{\frac{T_3}{k}}\\
        u_4 = \theta_1 = \pi/2 - \delta\\
        u_5 = \theta_2 = \pi/2 + \delta\\
        u_6 = \theta_3 = \theta_{3,eq}\\
        u_7 = \theta_4 = -\pi/2
    \end{gather}
\end{frame}

        \begin{column}{0.48\textwidth}
            \begin{block}{Controllo Quota ($z$) $\rightarrow$ Pitch ($\theta_{des}$)}
                Il controllore di quota $z$ calcola un angolo di beccheggio desiderato:
                % \textbf{Generazione Riferimento:}
                \begin{equation}
                    \theta_{des} = K_{P_{z}} e_z + K_{I_{z}} \int_{0}^{t} e_z(\tau) d\tau + K_{D_{z}} \dot{e}_z
                \end{equation}
                con $e_z = z_{des} - z$.             
            \end{block}
        \end{column}

        % Colonna Velocità
        \begin{column}{0.48\textwidth}
            \begin{block}{Controllo Velocità ($v_x$) $\rightarrow$ Spinta ($F_{x}$)}
                Il controllore di velocità genera una forza propulsiva lungo $x$:
                % \textbf{Forza Richiesta:}
                \begin{equation}
                    F_{x,req} = \hat{D}_x(v) + K_{P_v} e_v + K_{I_v} \int_{0}^{t} e_v(\tau) d\tau
                \end{equation}
                con $e_v = v_{x,des} - v_x$.
            \end{block}
        \end{column}
    \end{columns}

    % \vspace{0.1cm}
    \begin{columns}[c] 
        \begin{column}{0.7\textwidth}
            \begin{block}{Controllo Laterale ($y$) $\rightarrow$ Yaw ($\psi_{des}$)}
                Il controllore laterale calcola un angolo di imbardata di riferimento:
                \begin{equation}
                    \psi_{des} = K_{P_y} e_y + K_{D_y} \dot{e}_y
                \end{equation}
                con $e_y = y_{des} - y$
            \end{block}
        \end{column}
    \end{columns}

\end{frame}




% --- SLIDE 4: INNER LOOP (ANALOGIE) ---
\begin{frame}{Inner Loop: Thrust Vectoring come Superfici Virtuali}
    In assenza di superfici aerodinamiche mobili, il MMA emula le superfici di controllo classiche usando i motori.\\
    I controllori sono dei \textbf{Proporzionali-Derivativi}. 

    \vspace{0.15cm}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{alertblock}{Pitch ($\theta$)}
                \centering \textbf{"Elevatori"}
                \vspace{0.2cm}
                \begin{itemize}
                    \item \textbf{Azione:} Tilt Collettivo.
                    \item $\theta_1 = \theta_2$.
                    \item Ruota il vettore spinta su/giù.
                \end{itemize}
                \begin{equation}
                    M_y = K_{P_\theta} e_\theta - K_{D_\theta} \dot{e}_{\theta}
                \end{equation}
                
            \end{alertblock}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{block}{Roll ($\phi$)}
                \centering \textbf{"Alettoni"}
                \vspace{0.2cm}
                \begin{itemize}
                    \item \textbf{Azione:} Tilt Differenziale.
                    \item $\theta_1 = -\theta_2$.
                    \item Crea coppia torcente sull'asse $X$.
                \end{itemize}
                \begin{equation}
                    M_x = K_{P_\phi} e_\phi - K_{D_\phi} \dot{e}_{\phi}
                \end{equation}
            \end{block}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{exampleblock}{Yaw ($\psi$)}
                \centering \textbf{"Timone"}
                \vspace{0.2cm}
                \begin{itemize}
                    \item \textbf{Azione:} Spinta Differenziale.
                    \item $T_1 \neq T_2$.
                    \item Crea coppia torcente sull'asse $Z$.
                \end{itemize}
                \begin{equation}
                    M_z = K_{P_\psi} e_\psi - K_{D_\psi} \dot{e}_{\psi}
                \end{equation}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}


% --- SLIDE 5: MMA CONCETTO (SINTESI VETTORIALE) ---
\begin{frame}{Motor Mixing Algorithm}

    \begin{itemize}
        \item \textbf{Thrust:}
            La spinta totale è determinata dalla richiesta del loop di velocità longitudinale.
            \begin{equation}
                T_{tot} = F_{x,req} 
            \end{equation}
        \item \textbf{Beccheggio:} si definisce $\alpha$ come l'inclinazione collettiva dei rotori e si assumono piccoli angoli di inclinazione
            \begin{equation}
                M_y = d_{mx} T_{tot} \sin(\alpha)
                \implies \alpha = \frac{M_{y,req}}{T_{tot} d_{mx}}
            \end{equation}

        \item \textbf{Rollio:} si definisce $\delta$ come l'inclinazione differenziale dei rotori e si assumono piccoli angoli di inclinazione
            \begin{equation}
                M_x = d_{my} T_{tot} \sin(\delta)
                \implies \delta = \frac{M_{x,req}}{T_{tot} d_{my}}
            \end{equation}

        \item \textbf{Imbardata:} si definisce la spinta differenziale $\Delta T = T_2 - T_1$
            \begin{equation}
                M_z = d_{my} (T_2 - T_1)
                \implies \Delta T = \frac{M_z}{d_{my}}
            \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Ingressi di controllo}
    I comandi dati agli attuatori sono quindi i seguenti:
\begin{gather}
    u_1 = \omega_1 = \sqrt{\frac{1}{k} \left(\frac{F_{x,req}}{2} - \frac{\Delta T}{2} \right)}\\
    u_2 = \omega_2 = \sqrt{\frac{1}{k} \left(\frac{F_{x,req}}{2} + \frac{\Delta T}{2} \right)}\\
    u_3 = \omega_3 = 0\\
    u_4 = \theta_1 = \alpha - \delta\\
    u_5 = \theta_2 = \alpha + \delta\\
    u_6 = \theta_3 = 0\\
    u_7 = \theta_4 = 0
\end{gather}
\end{frame}

% SLIDE 1: METODOLOGIA
\begin{frame}{Metodologia di Linearizzazione}
    L'analisi della stabilità locale è condotta tramite la \textbf{linearizzazione numerica differenziale} del sistema ad anello chiuso:
    \begin{equation}
        \dot{x} = f_{cl}(x) = f(x, u(x))
    \end{equation}

    \begin{block}{Regolarizzazione del Modello ($f_{cl} \in C^1$)}
        Per garantire la differenziabilità nel punto di equilibrio $x_{eq}$:
        \begin{itemize}
            \item \textbf{Rimozione saturazioni:} Ipotesi di piccoli segnali ($\delta x$ infinitesimo).
            \item \textbf{Approssimazione Sliding Mode:} $K \tanh(s/\Phi) \approx \frac{K}{\Phi}s$.
        \end{itemize}
    \end{block}

    \begin{block}{Differenze Finite Centrali}
        Calcolo dello Jacobiano $A_{cl}$ con accuratezza $\mathcal{O}(h^2)$ e passo $h=10^{-6}$:
        \begin{equation}
            A_{ij} = \frac{\partial f_i}{\partial x_j} \approx \frac{f_i(x_{eq} + h e_j) - f_i(x_{eq} - h e_j)}{2h}
        \end{equation}
    \end{block}
\end{frame}
