\section*{1. Cinematica della Posizione}
Relazione tra le velocità nel sistema corpo ($x_4, x_5, x_6$) e le velocità nel sistema inerziale ($\dot{x}_1, \dot{x}_2, \dot{x}_3$).

\begin{align}
\dot{x}_1 &= x_6 (\sin x_7 \sin x_9 + \cos x_7 \cos x_9 \sin x_8) \notag \\
&\quad - x_5 (\cos x_7 \sin x_9 - \cos x_9 \sin x_7 \sin x_8) + x_4 \cos x_8 \cos x_9 \\[1em]
\dot{x}_2 &= x_5 (\cos x_7 \cos x_9 + \sin x_7 \sin x_8 \sin x_9) \notag \\
&\quad - x_6 (\cos x_9 \sin x_7 - \cos x_7 \sin x_8 \sin x_9) + x_4 \cos x_8 \sin x_9 \\[1em]
\dot{x}_3 &= x_6 \cos x_7 \cos x_8 - x_4 \sin x_8 + x_5 \cos x_8 \sin x_7
\end{align}

\section*{2. Dinamica Traslazionale}
Equazioni delle forze espresse nel sistema corpo. Notare l'aggiunta dei termini di tilt dei rotori ($\cos x_{13}, \cos x_{15}, \dots$).

\begin{align}
\dot{x}_4 &= -\frac{1}{m} \bigg( g m \sin x_8 - k x_{21}^2 \cos x_{13} - k x_{23}^2 \cos x_{15} - m x_5 x_{12} + m x_6 x_{11} \notag \\
&\quad - k x_{25}^2 \cos x_{17} \cos x_{19} + C_d \rho s x_4^2 \sgn(x_4) + \frac{1}{2} C_{d_x} \rho s_{body\_x} x_4^2 \sgn(x_4) \bigg) \\[1em]
\dot{x}_5 &= -\frac{1}{m} \bigg( k x_{25}^2 \cos x_{17} \sin x_{19} - m x_4 x_{12} + m x_6 x_{10} \notag \\
&\quad + g m \cos x_8 \sin x_7 - \frac{1}{2} C_{d_y} \rho s_{body\_y} x_5^2 \sgn(x_5) \bigg) \\[1em]
\dot{x}_6 &= -\frac{1}{m} \bigg( k x_{21}^2 \sin x_{13} + k x_{23}^2 \sin x_{15} + k x_{25}^2 \sin x_{17} - m x_4 x_{11} + m x_5 x_{10} \notag \\
&\quad + C_l \rho s x_4^2 - g m \cos x_7 \cos x_8 + \frac{1}{2} C_{d_z} \rho s_{body\_z} x_6^2 \sgn(x_6) \bigg)
\end{align}

\section*{3. Cinematica dell'Orientamento}
Relazione tra le velocità angolari del corpo e le derivate degli angoli di Eulero (invariata).

\begin{align}
\dot{x}_7 &= x_{10} + x_{12} \cos x_7 \tan x_8 + x_{11} \sin x_7 \tan x_8 \\[1em]
\dot{x}_8 &= x_{11} \cos x_7 - x_{12} \sin x_7 \\[1em]
\dot{x}_9 &= \frac{x_{12} \cos x_7}{\cos x_8} + \frac{x_{11} \sin x_7}{\cos x_8}
\end{align}

\section*{4. Dinamica Rotazionale}
Equazioni dei momenti. Queste equazioni ora includono i contributi inerziali e giroscopici dei rotori basculanti e le relative velocità di tilt ($x_{14}, x_{16}, x_{18}, x_{20}$).

\begin{align}
\dot{x}_{10} &= -\frac{1}{I_{xx}} \bigg( x_{10}(x_5^2 + 1) - b x_{21}^2 \cos x_{13} + b x_{23}^2 \cos x_{15} - I_{yy} x_{11} x_{12} + I_{zz} x_{11} x_{12} \notag \\
&\quad - b x_{25}^2 \cos x_{17} \cos x_{19} + I_{rotor\_zz} x_{14} x_{21} \sin x_{13} - I_{rotor\_zz} x_{16} x_{23} \sin x_{15} \notag \\
&\quad + I_{rotor\_zz} x_{18} x_{25} \sin x_{17} + d_{my} k x_{21}^2 \sin x_{13} - d_{my} k x_{23}^2 \sin x_{15} \notag \\
&\quad + I_{rotor\_yy} x_{20} x_{25} \cos x_{17} \sin x_{19} \bigg) \\[1em]
\dot{x}_{11} &= \frac{1}{I_{yy}} \bigg( I_{zz} x_{10} x_{12} - I_{xx} x_{10} x_{12} + b x_{25}^2 \cos x_{17} \sin x_{19} + d_{mx} k x_{21}^2 \sin x_{13} \notag \\
&\quad + d_{mx} k x_{23}^2 \sin x_{15} + d_{tx} k x_{25}^2 \sin x_{17} + x_{20} x_{25} \cos x_{17} \cos x_{19}(m_{rotor} d_{tx}^2 + I_{rotor\_xx}) \bigg) \\[1em]
\dot{x}_{12} &= -\frac{1}{I_{zz}} \bigg( x_{12}(x_5^2 + 1) + b x_{21}^2 \sin x_{13} - b x_{23}^2 \sin x_{15} + b x_{25}^2 \sin x_{17} \notag \\
&\quad - I_{xx} x_{10} x_{11} + I_{yy} x_{10} x_{11} + x_{14} x_{21} \cos x_{13}(m_{rotor} d_{mx}^2 + I_{rotor\_xx}) \notag \\
&\quad - x_{16} x_{23} \cos x_{15}(m_{rotor} d_{mx}^2 + I_{rotor\_xx}) + d_{my} k x_{21}^2 \cos x_{13} - d_{my} k x_{23}^2 \cos x_{15} \notag \\
&\quad + x_{18} x_{25} \cos x_{17} \cos x_{19}(m_{rotor} d_{tx}^2 + I_{rotor\_xx}) - d_{tx} k x_{25}^2 \cos x_{17} \sin x_{19} \bigg)
\end{align}
